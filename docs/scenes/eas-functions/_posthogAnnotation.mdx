import { GithubIcon } from '@expo/styleguide-icons/custom/GithubIcon';

import { BoxLink } from '~/ui/components/BoxLink';
import { YesIcon, NoIcon } from '~/ui/components/DocIcons';

| 属性           | 类型    | 必填       | 描述                                                                                                                                                         |
| -------------- | ------- | ---------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| `content`      | string  | <YesIcon /> | 注释文本。                                                                                                                                                   |
| `date_marker`  | string  | <NoIcon />  | 注释固定到的 ISO 8601 时间戳。默认为当前时间。                                                                                                                 |
| `api_key`      | string  | <NoIcon />  | PostHog 个人 API 密钥。默认为 `POSTHOG_CLI_API_KEY` 环境变量。需要 `annotation:write` 作用域。                                                               |
| `project_id`   | string  | <NoIcon />  | PostHog 项目 ID。默认为 `POSTHOG_CLI_PROJECT_ID` 环境变量。                                                                                                  |
| `ignore_error` | boolean | <NoIcon />  | 当值为 `true` 时，会记录网络错误或意外响应，但不会使步骤失败。默认为 `false`。权限错误始终会使步骤失败。                                                     |
