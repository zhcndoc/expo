import { GithubIcon } from '@expo/styleguide-icons/custom/GithubIcon';

import { BoxLink } from '~/ui/components/BoxLink';
import { YesIcon, NoIcon } from '~/ui/components/DocIcons';

| 属性               | 类型   | 必填       | 描述                                                                                                             |
| ------------------ | ------ | ---------- | ---------------------------------------------------------------------------------------------------------------- |
| `query`            | string | <YesIcon /> | HogQL 查询。第一行的第一列必须是单个数字。                                                                       |
| `operator`         | string | <YesIcon /> | 比较运算符。可以是 `lt`、`lte`、`gt`、`gte` 或 `eq` 之一。当 `value <operator> threshold` 成立时，步骤将清除。 |
| `threshold`        | number | <YesIcon /> | 用于与查询结果进行比较的值。                                                                                     |
| `timeout_seconds`  | number | <NoIcon />  | 等待的最长时间，以秒为单位。默认为 `600`。                                                                       |
| `interval_seconds` | number | <NoIcon />  | 检查之间的时间间隔，以秒为单位。默认为 `30`。                                                                    |
| `api_key`          | string | <NoIcon />  | PostHog 个人 API 密钥。默认为 `POSTHOG_CLI_API_KEY` 环境变量。需要 `query:read` 权限范围。                      |
| `project_id`       | string | <NoIcon />  | PostHog 项目 ID。默认为 `POSTHOG_CLI_PROJECT_ID` 环境变量。                                                       |

##### 输出

| 属性    | 类型   | 描述                           |
| ------- | ------ | ------------------------------ |
| `value` | string | 满足比较条件的指标值。         |
