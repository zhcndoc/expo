Expo UI LazyHStack 与官方 SwiftUI [LazyHStack API](https://developer.apple.com/documentation/swiftui/lazyhstack) 保持一致，并水平排列其子项，仅在需要时创建项目（当它们在滚动过程中变为可见时）。

> **info** `LazyHStack` 目前还不是真正的懒加载：原生端只会构建可见项，但 React 仍会一次性创建所有子项，因此大型列表的挂载可能会很慢。我们正在改进这一点。对于大型列表，我们建议使用 [FlashList](https://shopify.github.io/flash-list/) 或 [Legend List](https://github.com/LegendApp/legend-list)。

## 用法

### 基本的懒加载水平堆栈

LazyHStack 应该放在带有 `axes="horizontal"` 的 `ScrollView` 内，以启用懒加载渲染。

<ComponentExample
  title="BasicLazyHStackExample.tsx"
  src="/static/images/expo-ui/examples/lazyhstack-basic-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/lazyhstack-basic-ios-dark.webp"
  alt="屏幕中央水平排列着编号从零到五的项目。">

```tsx
import {
  Host,
  ScrollView,
  LazyHStack,
  Text,
} from '@expo/ui/swift-ui';

export default function BasicLazyHStackExample() {
  return (
    <Host style={{ flex: 1 }}>
      <ScrollView axes="horizontal">
        <LazyHStack spacing={12}>
          {Array.from({ length: 100 }, (_, i) => (
            <Text key={i}>{`项目 ${i}`}</Text>
          ))}
        </LazyHStack>
      </ScrollView>
    </Host>
  );
}
```

</ComponentExample>

### 设置对齐方式

`alignment` 属性控制子项的垂直对齐方式。可用选项有：`top`、`center`、`bottom`、`firstTextBaseline` 和 `lastTextBaseline`。

<ComponentExample
  title="LazyHStackAlignmentExample.tsx"
  src="/static/images/expo-ui/examples/lazyhstack-alignment-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/lazyhstack-alignment-ios-dark.webp"
  alt="一行中有三个高度不同的矩形，沿顶部边缘对齐。">

```tsx
import {
  Host,
  ScrollView,
  LazyHStack,
  Rectangle,
} from '@expo/ui/swift-ui';
import { frame } from '@expo/ui/swift-ui/modifiers';

export default function LazyHStackAlignmentExample() {
  return (
    <Host
      matchContents={{ horizontal: true }}
      style={{ flex: 1, alignSelf: 'center' }}>
      <ScrollView axes="horizontal">
        <LazyHStack spacing={12} alignment="top">
          <Rectangle
            modifiers={[frame({ width: 50, height: 50 })]}
          />
          <Rectangle
            modifiers={[frame({ width: 50, height: 100 })]}
          />
          <Rectangle
            modifiers={[frame({ width: 50, height: 75 })]}
          />
        </LazyHStack>
      </ScrollView>
    </Host>
  );
}
```

</ComponentExample>
