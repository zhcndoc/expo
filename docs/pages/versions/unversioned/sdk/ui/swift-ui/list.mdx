> **info** 有关跨平台用法，请参阅通用的 [`List`](../universal/list) — 它会为每个平台渲染相应的原生组件。

Expo UI List 与官方 SwiftUI [List API](https://developer.apple.com/documentation/swiftui/list) 保持一致，并支持通过 [`listStyle`](modifiers/#liststylestyle) 修饰符、各种行/分区修饰符，以及选择、重新排序和编辑功能进行样式设置。

> **info** `List` 目前还不会延迟渲染各行：React 会预先创建每一行，因此大型列表在挂载时可能会很慢。我们正在改进这一点。对于大型列表，我们推荐使用 [FlashList](https://shopify.github.io/flash-list/) 或 [Legend List](https://github.com/LegendApp/legend-list)。

### 列表样式

使用 [`listStyle`](modifiers/#liststylestyle) 修饰符来更改列表的外观。

> **注意：** `inset`、`insetGrouped` 和 `sidebar` 样式在 tvOS 上不可用。

<ComponentExample
  title="ListStylesExample.tsx"
  src="/static/images/expo-ui/examples/list-styles-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/list-styles-ios-dark.webp"
  alt="一个列表，其中包含一个设置为 automatic 的 List Style 菜单选择器和三个示例项目">

```tsx
import { useState } from 'react';
import {
  Host,
  List,
  Text,
  Section,
  Picker,
} from '@expo/ui/swift-ui';
import {
  listStyle,
  pickerStyle,
  tag,
} from '@expo/ui/swift-ui/modifiers';

const styles = [
  'automatic',
  'plain',
  'inset',
  'insetGrouped',
  'grouped',
  'sidebar',
] as const;

export default function ListStylesExample() {
  const [styleIndex, setStyleIndex] = useState(0);

  return (
    <Host style={{ flex: 1 }}>
      <List modifiers={[listStyle(styles[styleIndex])]}>
        <Section title="样式选择器">
          <Picker
            label="列表样式"
            selection={styleIndex}
            onSelectionChange={setStyleIndex}
            modifiers={[pickerStyle('menu')]}>
            {styles.map((style, index) => (
              <Text key={style} modifiers={[tag(index)]}>
                {style}
              </Text>
            ))}
          </Picker>
        </Section>
        <Section title="示例项目">
          <Text>项目 1</Text>
          <Text>项目 2</Text>
          <Text>项目 3</Text>
        </Section>
      </List>
    </Host>
  );
}
```

</ComponentExample>

### 选择和编辑模式

使用带有 `onDelete` 和 `onMove` 属性的 [`List.ForEach`](#listforeach) 复合组件，可启用列表项的选择、删除和重新排序。

- 使用 [`environment`](modifiers/#environmentkey-value) 修饰符启用编辑模式
- 使用 `keyExtractor` 属性标识项目
- 使用 [`selection`](#selection) 属性控制选中的项目
- 使用 [`moveDisabled`](modifiers/#movedisableddisabled) 和 [`deleteDisabled`](modifiers/#deletedisableddisabled) 修饰符禁用单个项目上的这些操作

<ComponentExample
  title="EditableListExample.tsx"
  src="/static/images/expo-ui/examples/list-editable-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/list-editable-ios-dark.webp"
  alt="包含编辑模式切换开关和四个任务行的列表">

```tsx
import { useState } from 'react';
import {
  Host,
  List,
  Label,
  Section,
  Button,
  Toggle,
} from '@expo/ui/swift-ui';
import { environment } from '@expo/ui/swift-ui/modifiers';

type Task = { id: string; title: string };

const INITIAL_TASKS: Task[] = [
  { id: '1', title: '任务 1' },
  { id: '2', title: '任务 2' },
  { id: '3', title: '任务 3' },
  { id: '4', title: '任务 4' },
];

export default function EditableListExample() {
  const [tasks, setTasks] = useState<Task[]>(INITIAL_TASKS);
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  const [editMode, setEditMode] = useState(false);

  const handleDelete = (indices: number[]) => {
    setTasks(prev => prev.filter((_, i) => !indices.includes(i)));
  };

  const handleMove = (
    sourceIndices: number[],
    destination: number
  ) => {
    setTasks(prev => {
      const newTasks = [...prev];
      const [removed] = newTasks.splice(sourceIndices[0], 1);
      const adjustedDest =
        sourceIndices[0] < destination
          ? destination - 1
          : destination;
      newTasks.splice(adjustedDest, 0, removed);
      return newTasks;
    });
  };

  return (
    <Host style={{ flex: 1 }}>
      <List
        selection={selectedIds}
        onSelectionChange={ids => setSelectedIds(ids.map(String))}
        modifiers={[
          environment('editMode', editMode ? 'active' : 'inactive'),
        ]}>
        <Section title="设置">
          <Toggle
            label="编辑模式"
            isOn={editMode}
            onIsOnChange={setEditMode}
          />
        </Section>
        <Section title="任务">
          <List.ForEach
            data={tasks}
            keyExtractor={task => task.id}
            onDelete={handleDelete}
            onMove={handleMove}>
            {({ item }) => <Label title={item.title} />}
          </List.ForEach>
        </Section>
      </List>
    </Host>
  );
}
```

</ComponentExample>

### 下拉刷新

使用 [`refreshable`](modifiers/#refreshablehandler) 修饰符启用下拉刷新功能。

<ComponentExample
  title="RefreshableListExample.tsx"
  src="/static/images/expo-ui/examples/list-refreshable-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/list-refreshable-ios-dark.webp"
  alt="包含下拉刷新行的列表">

```tsx
import { useState } from 'react';
import { Host, List, Text, Section } from '@expo/ui/swift-ui';
import { refreshable } from '@expo/ui/swift-ui/modifiers';

export default function RefreshableListExample() {
  const [lastRefresh, setLastRefresh] = useState<Date | null>(null);

  const handleRefresh = async () => {
    // 模拟异步数据获取
    await new Promise(resolve => setTimeout(resolve, 1500));
    setLastRefresh(new Date());
  };

  return (
    <Host style={{ flex: 1 }}>
      <List modifiers={[refreshable(handleRefresh)]}>
        <Section title="数据">
          <Text>下拉以刷新</Text>
          {lastRefresh && (
            <Text>
              上次刷新：{lastRefresh.toLocaleTimeString()}
            </Text>
          )}
        </Section>
      </List>
    </Host>
  );
}
```

</ComponentExample>

### 行样式

使用 [`listRowBackground`](modifiers/#listrowbackgroundcolor)、[`listRowSeparator`](modifiers/#listrowseparatorvisibility-edges)、[`listRowSeparatorTint`](modifiers/#listrowseparatortintcolor-edges) 和 [`listRowInsets`](modifiers/#listrowinsetsparams) 修饰符自定义单独的行。将 [`alignmentGuide`](modifiers/#alignmentguideguide-value) 与 `listRowSeparatorLeading` 对齐指南搭配使用，可设置行分隔线的起始位置。

<ComponentExample
  title="RowStylingExample.tsx"
  src="/static/images/expo-ui/examples/list-row-styling-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/list-row-styling-ios-dark.webp"
  alt="一个分组列表，其中包含蓝色行、红色分隔线、缩进的行以及一行与边缘齐平">

```tsx
import { Host, List, Text, Section } from '@expo/ui/swift-ui';
import {
  alignmentGuide,
  listRowBackground,
  listRowSeparator,
  listRowSeparatorTint,
  listRowInsets,
} from '@expo/ui/swift-ui/modifiers';

export default function RowStylingExample() {
  return (
    <Host style={{ flex: 1 }}>
      <List>
        <Section title="已设置样式的行">
          <Text modifiers={[listRowBackground('blue')]}>
            蓝色背景
          </Text>
          <Text modifiers={[listRowSeparator('hidden')]}>
            隐藏分隔线
          </Text>
          <Text modifiers={[listRowSeparatorTint('red')]}>
            红色分隔线
          </Text>
          <Text modifiers={[listRowInsets({ leading: 40 })]}>
            增加前导内边距
          </Text>
          <Text
            modifiers={[
              listRowInsets({ leading: 0, trailing: 0 }),
            ]}>
            无水平内边距
          </Text>
          <Text
            modifiers={[
              alignmentGuide('listRowSeparatorLeading', 32),
            ]}>
            分隔线从距边缘 32 点处开始
          </Text>
        </Section>
      </List>
    </Host>
  );
}
```

</ComponentExample>
