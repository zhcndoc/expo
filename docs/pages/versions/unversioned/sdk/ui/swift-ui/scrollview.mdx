Expo UI ScrollView 与官方 SwiftUI [ScrollView API](https://developer.apple.com/documentation/swiftui/scrollview) 保持一致，并为其子元素提供一个可滚动容器。

## 用法

### 基本垂直滚动视图

一个简单的可垂直滚动文本项列表。

<ComponentExample
  title="ScrollViewVerticalExample.tsx"
  src="/static/images/expo-ui/examples/scrollview-vertical-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/scrollview-vertical-ios-dark.webp"
  alt="一个垂直滚动视图，列出从一到二十九的项目。">

```tsx
import { Host, ScrollView, VStack, Text } from '@expo/ui/swift-ui';
import { padding } from '@expo/ui/swift-ui/modifiers';

export default function ScrollViewVerticalExample() {
  return (
    <Host style={{ flex: 1 }}>
      <ScrollView>
        <VStack spacing={8}>
          {Array.from({ length: 30 }, (_, i) => (
            <Text key={i} modifiers={[padding({ horizontal: 16 })]}>
              {`项目 ${i + 1}`}
            </Text>
          ))}
        </VStack>
      </ScrollView>
    </Host>
  );
}
```

</ComponentExample>

### 水平滚动视图

使用 `axes` 属性进行水平滚动。

<ComponentExample
  title="ScrollViewHorizontalExample.tsx"
  src="/static/images/expo-ui/examples/scrollview-horizontal-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/scrollview-horizontal-ios-dark.webp"
  alt="一排圆角方块，颜色从红色渐变到橙色再到黄色。">

```tsx
import {
  Host,
  ScrollView,
  HStack,
  RoundedRectangle,
} from '@expo/ui/swift-ui';
import {
  frame,
  foregroundStyle,
} from '@expo/ui/swift-ui/modifiers';

export default function ScrollViewHorizontalExample() {
  return (
    <Host style={{ flex: 1 }}>
      <ScrollView axes="horizontal">
        <HStack spacing={8}>
          {Array.from({ length: 20 }, (_, i) => (
            <RoundedRectangle
              key={i}
              cornerRadius={12}
              modifiers={[
                frame({ width: 100, height: 100 }),
                foregroundStyle(`hsl(${i * 18}, 70%, 50%)`),
              ]}
            />
          ))}
        </HStack>
      </ScrollView>
    </Host>
  );
}
```

</ComponentExample>

### 共享滚动位置

> **info** 需要 iOS 17 或更高版本。在较旧的版本上，该修饰符不会产生任何作用。

从 JavaScript 跟踪领先的滚动目标 id，并通过写入状态来滚动到某个目标。为每个滚动目标添加 `id` 修饰符，将内容容器包裹在 `scrollTargetLayout` 中，并将 `scrollPosition` 修饰符应用到 `ScrollView`。可选的 `onChange` 回调会在领先目标发生变化时在 JS 线程上触发。

`scrollPosition` 修饰符也适用于其他可滚动容器，例如 `LazyVStack` 和 `LazyHStack`。

> **warning** 对 `state.value` 的写入必须在 UI 运行时中执行。请将写入操作包裹在 `react-native-worklets` 中的 `scheduleOnUI` 里，或者从 `'worklet'` 函数内部调用它们。来自 JS 运行时的写入会触发 Main Thread Checker，这是 Xcode 的运行时工具，用于标记在后台线程中进行的 UIKit 调用。

<ComponentExample
  title="ScrollViewSharedPositionExample.tsx"
  src="/static/images/expo-ui/examples/scrollview-shared-position-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/scrollview-shared-position-ios-dark.webp"
  alt="一个可滚动项目列表，列表上方有一个标记为“从 worklet 滚动到 item 10”的按钮。">

```tsx
import {
  Button,
  Host,
  ScrollView,
  Text,
  VStack,
  useNativeState,
} from '@expo/ui/swift-ui';
import {
  id,
  padding,
  scrollPosition,
  scrollTargetLayout,
} from '@expo/ui/swift-ui/modifiers';
import { scheduleOnUI } from 'react-native-worklets';

export default function ScrollViewSharedPositionExample() {
  const activeID = useNativeState<string | null>(null);

  return (
    <Host style={{ flex: 1 }}>
      <VStack spacing={12}>
        <ScrollView
          modifiers={[
            scrollPosition(activeID, {
              onChange: newID => {
                console.log('[JS thread] leading target:', newID);
              },
            }),
          ]}>
          <VStack modifiers={[scrollTargetLayout()]}>
            {Array.from({ length: 30 }, (_, i) => (
              <Text
                key={`item-${i}`}
                modifiers={[
                  id(`item-${i}`),
                  padding({ horizontal: 16, vertical: 12 }),
                ]}>
                {`Item ${i}`}
              </Text>
            ))}
          </VStack>
        </ScrollView>
        <Button
          label="从 worklet 滚动到 item 10"
          onPress={() => {
            scheduleOnUI(() => {
              'worklet';
              activeID.value = 'item-10';
            });
          }}
        />
      </VStack>
    </Host>
  );
}
```

</ComponentExample>

### 未裁剪的滚动内容

> **info** 需要 iOS 17 或更高版本。在较旧的版本上，该修饰符不会产生任何作用。

滚动视图会将其内容裁剪到自身边界，超出边界绘制的内容（例如阴影或超出边缘放大的卡片）会被截断。应用 [`scrollClipDisabled`](modifiers/#scrollclipdisableddisabled) 修饰符以保持这些内容可见。

<ComponentExample
  title="ScrollViewUnclippedExample.tsx"
  src="/static/images/expo-ui/examples/scrollview-unclipped-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/scrollview-unclipped-ios-dark.webp"
  alt="一排彩色方块，其阴影延伸到滚动边界之外">

```tsx
import {
  Host,
  HStack,
  RoundedRectangle,
  ScrollView,
} from '@expo/ui/swift-ui';
import {
  frame,
  foregroundStyle,
  scrollClipDisabled,
  shadow,
} from '@expo/ui/swift-ui/modifiers';

export default function ScrollViewUnclippedExample() {
  return (
    <Host style={{ flex: 1 }}>
      <ScrollView
        axes="horizontal"
        modifiers={[scrollClipDisabled()]}>
        <HStack>
          {Array.from({ length: 20 }, (_, i) => (
            <RoundedRectangle
              key={i}
              cornerRadius={12}
              modifiers={[
                frame({ width: 100, height: 100 }),
                foregroundStyle(`hsl(${i * 18}, 70%, 50%)`),
                shadow({ radius: 10, color: 'secondary' }),
              ]}
            />
          ))}
        </HStack>
      </ScrollView>
    </Host>
  );
}
```

</ComponentExample>
