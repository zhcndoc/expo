### 带分区的表单

使用 [`Section`](section) 组件在表单内对相关控件进行分组。

<ComponentExample
  title="FormWithSectionsExample.tsx"
  src="/static/images/expo-ui/examples/form-with-sections-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/form-with-sections-ios-dark.webp"
  alt="一个表单，包含一个有两个字段的个人资料分区、一个有两个切换开关的偏好设置分区，以及一个保存更改按钮">

```tsx
import { useState } from 'react';
import {
  Host,
  Form,
  Section,
  TextField,
  Toggle,
  Button,
} from '@expo/ui/swift-ui';

export default function FormWithSectionsExample() {
  const [notifications, setNotifications] = useState(false);
  const [darkMode, setDarkMode] = useState(false);

  return (
    <Host style={{ flex: 1 }}>
      <Form>
        <Section title="个人资料">
          <TextField placeholder="姓名" />
          <TextField placeholder="电子邮件" />
        </Section>

        <Section title="偏好设置">
          <Toggle
            label="启用通知"
            isOn={notifications}
            onIsOnChange={setNotifications}
          />
          <Toggle
            label="深色模式"
            isOn={darkMode}
            onIsOnChange={setDarkMode}
          />
        </Section>

        <Section>
          <Button
            label="保存更改"
            onPress={() => console.log('Saved!')}
          />
        </Section>
      </Form>
    </Host>
  );
}
```

</ComponentExample>

### 带自定义背景的表单

使用 `scrollContentBackground` 修饰器来自定义或隐藏表单的背景。

<ComponentExample
  title="FormBackgroundExample.tsx"
  src="/static/images/expo-ui/examples/form-background-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/form-background-ios-dark.webp"
  alt="一个表单，其自定义背景分区后方带有浅灰色背景">

```tsx
import { Host, Form, Section, TextField } from '@expo/ui/swift-ui';
import {
  scrollContentBackground,
  background,
} from '@expo/ui/swift-ui/modifiers';

export default function FormBackgroundExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Form
        modifiers={[
          scrollContentBackground('hidden'),
          background('#F0F0F0'),
        ]}>
        <Section title="自定义背景">
          <TextField placeholder="输入文本" />
        </Section>
      </Form>
    </Host>
  );
}
```

</ComponentExample>

### 不可滚动的表单

使用 [`scrollDisabled`](modifiers/#scrolldisableddisabled) 修饰器来阻止表单滚动。

> **注意：** `scrollDisabled` 修饰器仅适用于 iOS 16+ 和 tvOS 16+。

<ComponentExample
  title="NonScrollableFormExample.tsx"
  src="/static/images/expo-ui/examples/form-non-scrollable-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/form-non-scrollable-ios-dark.webp"
  alt="一个包含设置分区的表单，其中有一个启用功能切换开关">

```tsx
import { useState } from 'react';
import {
  Host,
  Form,
  Section,
  TextField,
  Toggle,
} from '@expo/ui/swift-ui';
import { scrollDisabled } from '@expo/ui/swift-ui/modifiers';

export default function NonScrollableFormExample() {
  const [isOn, setIsOn] = useState(false);

  return (
    <Host style={{ flex: 1 }}>
      <Form modifiers={[scrollDisabled()]}>
        <Section title="设置">
          <Toggle
            label="启用功能"
            isOn={isOn}
            onIsOnChange={setIsOn}
          />
        </Section>
      </Form>
    </Host>
  );
}
```

</ComponentExample>

### 下拉刷新表单

使用 `refreshable` 修饰器添加下拉刷新功能。

<ComponentExample
  title="RefreshableFormExample.tsx"
  src="/static/images/expo-ui/examples/form-refreshable-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/form-refreshable-ios-dark.webp"
  alt="一个包含下拉刷新分区的表单，其中显示上次刷新的时间">

```tsx
import { useState, useCallback } from 'react';
import { Host, Form, Section, Text } from '@expo/ui/swift-ui';
import { refreshable } from '@expo/ui/swift-ui/modifiers';

export default function RefreshableFormExample() {
  const [lastRefresh, setLastRefresh] = useState(new Date());

  const handleRefresh = useCallback(async () => {
    // 模拟网络请求
    await new Promise(resolve => setTimeout(resolve, 1500));
    setLastRefresh(new Date());
  }, []);

  return (
    <Host style={{ flex: 1 }}>
      <Form modifiers={[refreshable(handleRefresh)]}>
        <Section title="下拉刷新">
          <Text>
            上次刷新时间：{lastRefresh.toLocaleTimeString()}
          </Text>
        </Section>
      </Form>
    </Host>
  );
}
```

</ComponentExample>
