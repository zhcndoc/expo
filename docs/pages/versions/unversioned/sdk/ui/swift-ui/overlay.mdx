## 用法

<ComponentExample
  title="OverlayExample.tsx"
  src="/static/images/expo-ui/examples/overlay-basic-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/overlay-basic-ios-dark.webp"
  alt="一个蓝色铃铛图标，右上角叠加了一个红色圆形徽标，显示数字 3">

```tsx
import { Host, Overlay, Text, Image } from '@expo/ui/swift-ui';
import {
  foregroundStyle,
  frame,
  font,
  background,
  clipShape,
  offset,
} from '@expo/ui/swift-ui/modifiers';

export default function OverlayExample() {
  return (
    <Host matchContents>
      <Overlay alignment="topTrailing">
        <Image
          systemName="bell.fill"
          modifiers={[
            font({ size: 28 }),
            foregroundStyle('#007AFF'),
          ]}
        />
        <Overlay.Content>
          <Text
            modifiers={[
              font({ size: 11, weight: 'bold' }),
              foregroundStyle('#FFFFFF'),
              frame({ width: 18, height: 18 }),
              background('#FF3B30'),
              clipShape('circle'),
              offset({ x: 8, y: -8 }),
            ]}>
            3
          </Text>
        </Overlay.Content>
      </Overlay>
    </Host>
  );
}
```

</ComponentExample>
