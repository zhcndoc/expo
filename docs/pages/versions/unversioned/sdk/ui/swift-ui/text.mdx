<ComponentExample

### 带修饰符的文本

使用修饰符来样式化整段文本。

<ComponentExample
  title="StyledTextExample.tsx"
  src="/static/images/expo-ui/examples/text-styled-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-styled-ios-dark.webp"
  alt="以大号、粗体、蓝色字体显示的 Large Bold Blue Text">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';
import { font, foregroundStyle } from '@expo/ui/swift-ui/modifiers';

export default function StyledTextExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text
        modifiers={[
          font({ size: 24, weight: 'bold' }),
          foregroundStyle('blue'),
        ]}>
        Large Bold Blue Text
      </Text>
    </Host>
  );
}
```

</ComponentExample>

### 嵌套文本（按片段设置样式）

嵌套 `Text` 组件以对单个片段应用不同样式。这对于行内格式化非常有用，例如句子中的加粗或彩色词语。

> **注意：** 嵌套文本使用 SwiftUI 的 [Text 拼接](https://developer.apple.com/documentation/swiftui/text)，因此只有返回 `Text` 的修饰符（例如 `bold`、`italic`、`font`、`foregroundColor` 以及带颜色的 `foregroundStyle`）才会应用于嵌套片段。

<ComponentExample
  title="NestedTextExample.tsx"
  src="/static/images/expo-ui/examples/text-nested-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-nested-ios-dark.webp"
  alt="句子 Hello world!，其中 world 为粗体红色">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';
import {
  bold,
  italic,
  foregroundStyle,
} from '@expo/ui/swift-ui/modifiers';

export default function NestedTextExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text>
        Hello{' '}
        <Text modifiers={[bold(), foregroundStyle('red')]}>
          world
        </Text>
        !
      </Text>
    </Host>
  );
}
```

</ComponentExample>

### 混合行内样式

组合多个带样式的片段以实现富文本格式化。

<ComponentExample
  title="MixedStylesExample.tsx"
  src="/static/images/expo-ui/examples/text-mixed-styles-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-mixed-styles-ios-dark.webp"
  alt="一个句子，其中 bold 为粗体、italic 为斜体、colored 为橙色">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';
import {
  bold,
  italic,
  foregroundStyle,
  font,
} from '@expo/ui/swift-ui/modifiers';

export default function MixedStylesExample() {
  return (
    <Host matchContents>
      <Text>
        This is <Text modifiers={[bold()]}>bold</Text>,{' '}
        <Text modifiers={[italic()]}>italic</Text>, and{' '}
        <Text modifiers={[foregroundStyle('orange')]}>colored</Text>{' '}
        text.
      </Text>
    </Host>
  );
}
```

</ComponentExample>

### 字体字重

使用 `font` 修饰符来应用不同的字体字重。

<ComponentExample
  title="FontWeightsExample.tsx"
  src="/static/images/expo-ui/examples/text-font-weights-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-font-weights-ios-dark.webp"
  alt="八行堆叠文本，展示从 Ultra Light 到 Black 的字重">

```tsx
import { Host, Text, VStack } from '@expo/ui/swift-ui';
import { font } from '@expo/ui/swift-ui/modifiers';

export default function FontWeightsExample() {
  return (
    <Host style={{ flex: 1 }}>
      <VStack spacing={4}>
        <Text modifiers={[font({ weight: 'ultraLight' })]}>
          Ultra Light
        </Text>
        <Text modifiers={[font({ weight: 'light' })]}>Light</Text>
        <Text modifiers={[font({ weight: 'regular' })]}>
          Regular
        </Text>
        <Text modifiers={[font({ weight: 'medium' })]}>Medium</Text>
        <Text modifiers={[font({ weight: 'semibold' })]}>
          Semibold
        </Text>
        <Text modifiers={[font({ weight: 'bold' })]}>Bold</Text>
        <Text modifiers={[font({ weight: 'heavy' })]}>Heavy</Text>
        <Text modifiers={[font({ weight: 'black' })]}>Black</Text>
      </VStack>
    </Host>
  );
}
```

</ComponentExample>

### 自定义字体

使用带有 `family` 参数的 `font` 修饰符来使用自定义字体。你可以使用 [`expo-font`](/versions/latest/sdk/font/) 库加载自定义字体。

<ComponentExample
  title="CustomFontExample.tsx"
  src="/static/images/expo-ui/examples/text-custom-font-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-custom-font-ios-dark.webp"
  alt="两行文本，上方为 Inter Bold，下方为 Inter Regular。">

```tsx
import { Host, Text, VStack } from '@expo/ui/swift-ui';
import { font } from '@expo/ui/swift-ui/modifiers';

export default function CustomFontExample() {
  return (
    <Host matchContents style={{ alignSelf: 'center' }}>
      <VStack spacing={4}>
        <Text
          modifiers={[font({ family: 'Inter-Bold', size: 18 })]}>
          Inter Bold
        </Text>
        <Text
          modifiers={[font({ family: 'Inter-Regular', size: 18 })]}>
          Inter Regular
        </Text>
      </VStack>
    </Host>
  );
}
```

</ComponentExample>

### 带行数限制的文本

使用 `lineLimit` 修饰符在一定行数后截断文本。

<ComponentExample
  title="LineLimitExample.tsx"
  src="/static/images/expo-ui/examples/text-line-limit-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-line-limit-ios-dark.webp"
  alt="一段长文本在两行后截断，并以省略号结尾">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';
import { lineLimit } from '@expo/ui/swift-ui/modifiers';

export default function LineLimitExample() {
  const longText =
    'This is a very long text that will be truncated after two lines. '.repeat(
      5
    );

  return (
    <Host style={{ flex: 1 }}>
      <Text modifiers={[lineLimit(2)]}>{longText}</Text>
    </Host>
  );
}
```

</ComponentExample>

<ComponentExample

</ComponentExample>

### 自动更新的日期

使用 `date` 和 `dateStyle` 属性来显示会随着时间流逝自动更新的日期。这在小组件和实时活动中尤其有用。

<ComponentExample
  title="DateTextExample.tsx"
  src="/static/images/expo-ui/examples/text-date-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-date-ios-dark.webp"
  alt="显示 4 分 53 秒倒计时">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';

export default function DateTextExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text
        date={new Date(Date.now() + 300000)}
        dateStyle="timer"
      />
    </Host>
  );
}
```

</ComponentExample>

### 计时器区间

使用 `timerInterval` 来显示实时倒计时或正计时器。这需要 iOS/tvOS 16+。

<ComponentExample
  title="TimerIntervalExample.tsx"
  src="/static/images/expo-ui/examples/text-timer-interval-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-timer-interval-ios-dark.webp"
  alt="显示 9 分 54 秒倒计时">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';

export default function TimerIntervalExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text
        timerInterval={{
          lower: new Date(),
          upper: new Date(Date.now() + 600000),
        }}
        countsDown
      />
    </Host>
  );
}
```

</ComponentExample>

> **注意：** `timerInterval`、`countsDown` 和 `pauseTime` 需要 iOS 16.0+ / tvOS 16.0+。在较旧版本中，计时器区间将不会渲染。
