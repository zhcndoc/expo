Expo UI 的日期和时间选择器组件与官方 Jetpack Compose [Date Picker](https://developer.android.com/develop/ui/compose/components/datepickers)、[Date Range Picker](https://developer.android.com/develop/ui/compose/components/datepickers#range) 和 [Time Picker](https://developer.android.com/develop/ui/compose/components/time-pickers) API 相匹配。

> **注意：** 日期变体会渲染 Material 的日历网格和输入字段，这两者内部都会水平滚动。父级 `Host` 必须在水平方向提供有限宽度，请将 `matchContents={{ vertical: true }}` 与 `style={{ width: '100%' }}` 一起使用（或任何有限宽度）。详情请参阅 [Host 参考中的 Match contents](host/#match-contents)。

> **注意：** `DateRangePicker` 也会垂直滚动，并填满父级提供的有限高度。请将其 `Host` 放在有界布局中，例如使用 `flex: 1` 的布局。请勿在垂直轴上使用 `matchContents`。

## 用法

### 日期选择器

<ComponentExample
  title="DatePickerExample.tsx"
  src="/static/images/expo-ui/examples/datetimepicker-date-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/datetimepicker-date-android-dark.webp"
  alt="Material 3 日期选择器，显示 2026 年 8 月，并选中 14 日">

```tsx
import { useState } from 'react';
import { Host, DateTimePicker } from '@expo/ui/jetpack-compose';

export default function DatePickerExample() {
  const [selectedDate, setSelectedDate] = useState(new Date());

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <DateTimePicker
        onDateSelected={date => {
          setSelectedDate(date);
        }}
        displayedComponents="date"
        initialDate={selectedDate.toISOString()}
        variant="picker"
      />
    </Host>
  );
}
```

</ComponentExample>

### 日期范围选择器

<ComponentExample
  title="DateRangePickerAndroidExample.tsx"
  src="/static/images/expo-ui/examples/datetimepicker-date-range-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/datetimepicker-date-range-android-dark.webp"
  alt="日期范围选择器，已选中五天的范围，范围内的日期高亮显示">

```tsx
import { useState } from 'react';
import {
  DateRangePicker,
  type DateRangeSelection,
  Host,
} from '@expo/ui/jetpack-compose';

export default function DateRangePickerAndroidExample() {
  const [range, setRange] = useState<DateRangeSelection>({
    start: new Date(),
    end: new Date(Date.now() + 5 * 24 * 60 * 60 * 1000),
  });

  return (
    <Host style={{ flex: 1 }}>
      <DateRangePicker
        initialStartDate={range.start?.toISOString()}
        initialEndDate={range.end?.toISOString()}
        onDateRangeSelected={setRange}
      />
    </Host>
  );
}
```

</ComponentExample>

用户完成范围选择之前，回调中的 `end` 值为 `null`。

### 日期范围选择器对话框

<ComponentExample
  title="DateRangePickerDialogExample.tsx"
  src="/static/images/expo-ui/examples/datetimepicker-date-range-dialog-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/datetimepicker-date-range-dialog-android-dark.webp"
  alt="日期范围选择器模态对话框，起始日期和结束日期为空，并显示“取消”和“确定”按钮">

```tsx
import { useState } from 'react';
import { Button } from 'react-native';
import {
  DateRangePickerDialog,
  type DateRangeSelection,
  Host,
} from '@expo/ui/jetpack-compose';

export default function DateRangePickerDialogExample() {
  const [visible, setVisible] = useState(false);
  const [range, setRange] = useState<DateRangeSelection>({
    start: null,
    end: null,
  });

  return (
    <>
      <Button
        title="Select dates"
        onPress={() => setVisible(true)}
      />
      {visible && (
        <Host>
          <DateRangePickerDialog
            initialStartDate={range.start?.toISOString()}
            initialEndDate={range.end?.toISOString()}
            onDateRangeSelected={selectedRange => {
              setRange(selectedRange);
              setVisible(false);
            }}
            onDismissRequest={() => setVisible(false)}
          />
        </Host>
      )}
    </>
  );
}
```

</ComponentExample>
