## 用法

### 基础表面

<ComponentExample
  title="BasicSurfaceExample.tsx"
  src="/static/images/expo-ui/examples/surface-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/surface-basic-android-dark.webp"
  alt="一个浅色的 Material 3 表面面板，内含带有内边距的文本">

```tsx
import { Host, Surface, Text } from '@expo/ui/jetpack-compose';
import { paddingAll } from '@expo/ui/jetpack-compose/modifiers';

export default function BasicSurfaceExample() {
  return (
    <Host matchContents>
      <Surface>
        <Text modifiers={[paddingAll(16)]}>
          Content on a surface
        </Text>
      </Surface>
    </Host>
  );
}
```

</ComponentExample>

### 带 elevation 的表面

使用 `tonalElevation` 和 `shadowElevation` 来控制表面的视觉深度。

<ComponentExample
  title="SurfaceElevationExample.tsx"
  src="/static/images/expo-ui/examples/surface-elevation-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/surface-elevation-android-dark.webp"
  alt="两个叠放的表面，下方表面投下的阴影比上方表面更大">

```tsx
import {
  Host,
  Surface,
  Column,
  Text,
} from '@expo/ui/jetpack-compose';
import { paddingAll } from '@expo/ui/jetpack-compose/modifiers';

export default function SurfaceElevationExample() {
  return (
    <Host matchContents>
      <Column
        verticalArrangement={{ spacedBy: 16 }}
        modifiers={[paddingAll(16)]}>
        <Surface tonalElevation={1} shadowElevation={2}>
          <Text modifiers={[paddingAll(16)]}>Low elevation</Text>
        </Surface>
        <Surface tonalElevation={4} shadowElevation={8}>
          <Text modifiers={[paddingAll(16)]}>High elevation</Text>
        </Surface>
      </Column>
    </Host>
  );
}
```

</ComponentExample>

### 带自定义颜色的表面

使用 `color` 和 `contentColor` 属性覆盖默认的 Material 主题颜色。

<ComponentExample
  title="SurfaceCustomColorsExample.tsx"
  src="/static/images/expo-ui/examples/surface-colors-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/surface-colors-android-dark.webp"
  alt="一个深海军蓝色的表面面板，带有白色文本">

```tsx
import { Host, Surface, Text } from '@expo/ui/jetpack-compose';
import { paddingAll } from '@expo/ui/jetpack-compose/modifiers';

export default function SurfaceCustomColorsExample() {
  return (
    <Host matchContents>
      <Surface
        color="#1E3A5F"
        contentColor="#FFFFFF"
        tonalElevation={2}>
        <Text color="#FFFFFF" modifiers={[paddingAll(16)]}>
          Custom colored surface
        </Text>
      </Surface>
    </Host>
  );
}
```

</ComponentExample>

### 带形状和边框的表面

使用 `shape` 属性裁剪内容，并使用 `border` 在表面周围绘制描边。

<ComponentExample
  title="SurfaceShapeBorderExample.tsx"
  src="/static/images/expo-ui/examples/surface-shape-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/surface-shape-android-dark.webp"
  alt="一个带有 16dp 圆角的表面，周围有紫色的两像素边框">

```tsx
import {
  Host,
  Surface,
  Shape,
  Text,
} from '@expo/ui/jetpack-compose';
import { paddingAll } from '@expo/ui/jetpack-compose/modifiers';

export default function SurfaceShapeBorderExample() {
  return (
    <Host matchContents>
      <Surface
        shape={Shape.RoundedCorner({
          cornerRadii: {
            topStart: 16,
            topEnd: 16,
            bottomStart: 16,
            bottomEnd: 16,
          },
        })}
        border={{ width: 2, color: '#6200EE' }}>
        <Text modifiers={[paddingAll(16)]}>
          Rounded surface with border
        </Text>
      </Surface>
    </Host>
  );
}
```

</ComponentExample>
