一个水平布局容器，将其子元素从起始位置排列到结束位置。在 Android 上委托给 Jetpack Compose 的 [`Row`](../jetpack-compose/row)，在 iOS 上委托给 SwiftUI 的 [`HStack`](../swift-ui/hstack)，在 Web 上委托给 flex `View`。

```tsx
import { useColorScheme } from 'react-native';
import { Host, Row, Text } from '@expo/ui';

export default function RowExample() {
  const colorScheme = useColorScheme();
  const ink = { color: colorScheme === 'dark' ? '#FFFFFF' : '#000000' };

  return (
    <Host matchContents>
      <Row spacing={8}>
        <Text textStyle={ink}>One</Text>
        <Text textStyle={ink}>Two</Text>
        <Text textStyle={ink}>Three</Text>
      </Row>
    </Host>
  );
}
```

</ComponentExample>

```tsx
import { useColorScheme } from 'react-native';
import { Host, Row, Text } from '@expo/ui';

export default function RowAlignmentExample() {
  const colorScheme = useColorScheme();
  const ink = { color: colorScheme === 'dark' ? '#FFFFFF' : '#000000' };

  return (
    <Host style={{ width: '100%', height: 160 }}>
      <Row spacing={8} alignment="center">
        <Text textStyle={{ ...ink, fontSize: 30 }}>Large</Text>
        <Text textStyle={ink}>centered</Text>
      </Row>
    </Host>
  );
}
```

</ComponentExample>

### 使用 Spacer 推开内容

将 `Row` 与灵活的 [`Spacer`](spacer) 搭配使用，把其子元素推到相对两端。

<ComponentExample
  title="RowSpacerExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-row-spacer-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-row-spacer-android-dark.webp',
    alt: '首尾标签被推到相对两侧',
  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-row-spacer-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-row-spacer-ios-dark.webp',
    alt: '首尾标签被推到相对两侧',
  }}>

```tsx
import { useColorScheme } from 'react-native';
import { Host, Row, Text, Spacer } from '@expo/ui';

export default function RowSpacerExample() {
  const colorScheme = useColorScheme();
  const ink = { color: colorScheme === 'dark' ? '#FFFFFF' : '#000000' };

  return (
    <Host matchContents={{ vertical: true }} style={{ width: '100%' }}>
      <Row>
        <Text textStyle={ink}>Leading</Text>
        <Spacer flexible />
        <Text textStyle={ink}>Trailing</Text>
      </Row>
    </Host>
  );
}
```

</ComponentExample>
