## 日期选择器

<ComponentExample
  title="DatePickerExample.tsx"
  src="/static/images/expo-ui/examples/datepicker-date-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/datepicker-date-ios-dark.webp"
  alt="包含当前日期的紧凑按钮的“选择日期”行">

```tsx
import { useState } from 'react';
import { Host, DatePicker } from '@expo/ui/swift-ui';

export default function DatePickerExample() {
  const [selectedDate, setSelectedDate] = useState(new Date());

  // A date picker fills the width it is given, so `matchContents` collapses it.
  return (
    <Host style={{ flex: 1 }}>
      <DatePicker
        title="选择日期"
        selection={selectedDate}
        displayedComponents={['date']}
        onDateChange={date => {
          setSelectedDate(date);
        }}
      />
    </Host>
  );
}
```

</ComponentExample>

## 使用修饰符设置样式

你可以使用 `datePickerStyle` 修饰符更改选择器的外观。可用样式包括：`automatic`、`compact`、`graphical` 和 `wheel`。

<ComponentExample
  title="WheelDatePickerExample.tsx"
  src="/static/images/expo-ui/examples/datepicker-wheel-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/datepicker-wheel-ios-dark.webp"
  alt="包含日、月和年列的滚轮样式选择器">

```tsx
import { useState } from 'react';
import { Host, DatePicker } from '@expo/ui/swift-ui';
import { datePickerStyle } from '@expo/ui/swift-ui/modifiers';

export default function WheelDatePickerExample() {
  const [selectedDate, setSelectedDate] = useState(new Date());

  return (
    <Host style={{ flex: 1 }}>
      <DatePicker
        modifiers={[datePickerStyle('wheel')]}
        title="选择日期"
        selection={selectedDate}
        displayedComponents={['date']}
        onDateChange={date => {
          setSelectedDate(date);
        }}
      />
    </Host>
  );
}
```

</ComponentExample>

import { useState } from 'react';

  const [selectedDate, setSelectedDate] = useState(new Date());

  return (

</ComponentExample>

## 自定义区域设置

使用带有 `locale` 键的 `environment` 修饰符，可在特定区域设置下显示选择器。

<ComponentExample
  title="LocaleDatePickerExample.tsx"
  src="/static/images/expo-ui/examples/datepicker-locale-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/datepicker-locale-ios-dark.webp"
  alt="标签为 Sélectionner la date、以法语显示日期的选择器">

```tsx
import { useState } from 'react';
import { Host, DatePicker } from '@expo/ui/swift-ui';
import { environment } from '@expo/ui/swift-ui/modifiers';

export default function LocaleDatePickerExample() {
  const [selectedDate, setSelectedDate] = useState(new Date());

  return (
    <Host style={{ flex: 1 }}>
      <DatePicker
        title="选择日期"
        selection={selectedDate}
        displayedComponents={['date']}
        onDateChange={date => {
          setSelectedDate(date);
        }}
        modifiers={[environment('locale', 'fr_FR')]}
      />
    </Host>
  );
}
```

</ComponentExample>

## 自定义时区

使用带有 `timeZone` 键的 `environment` 修饰符，可在特定 IANA 时区中显示选择器。

<ComponentExample
  title="TimeZoneDatePickerExample.tsx"

  darkSrc="/static/images/expo-ui/examples/datepicker-timezone-ios-dark.webp"

```tsx
import { useState } from 'react';
import { Host, DatePicker } from '@expo/ui/swift-ui';
import { environment } from '@expo/ui/swift-ui/modifiers';

export default function TimeZoneDatePickerExample() {
  const [selectedDate, setSelectedDate] = useState(new Date());

  return (
    <Host style={{ flex: 1 }}>
      <DatePicker
        title="东京时间"
        selection={selectedDate}
        displayedComponents={['date', 'hourAndMinute']}

          setSelectedDate(date);

```

</ComponentExample>
