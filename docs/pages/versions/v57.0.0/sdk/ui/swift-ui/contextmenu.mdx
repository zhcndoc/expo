Expo UI ContextMenu 与官方 SwiftUI [contextMenu API](<https://developer.apple.com/documentation/swiftui/view/contextmenu(menuitems:)>) 保持一致，并会在长按时显示菜单。对于单击菜单交互，请改用 [`Menu`](menu)。

### 基础上下文菜单

<ComponentExample
  title="BasicContextMenuExample.tsx"
  src="/static/images/expo-ui/examples/contextmenu-basic-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/contextmenu-basic-ios-dark.webp"
  alt="触发器抬起后打开的上下文菜单，列出“编辑”和红色的“删除”">

```tsx
import { Host, ContextMenu, Button, Text } from '@expo/ui/swift-ui';

export default function BasicContextMenuExample() {
  return (
    <Host style={{ flex: 1 }}>
      <ContextMenu>
        <ContextMenu.Items>
          <Button
            label="Edit"
            onPress={() => console.log('Edit')}
          />
          <Button
            label="Delete"
            role="destructive"
            onPress={() => console.log('Delete')}
          />
        </ContextMenu.Items>
        <ContextMenu.Trigger>
          <Text>长按我</Text>
        </ContextMenu.Trigger>
      </ContextMenu>
    </Host>
  );
}
```

</ComponentExample>

### 带有预览的上下文菜单

使用 `ContextMenu.Preview` 可在打开菜单时在菜单上方显示自定义预览。

<ComponentExample
  title="ContextMenuWithPreviewExample.tsx"
  src="/static/images/expo-ui/examples/contextmenu-with-preview-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/contextmenu-with-preview-ios-dark.webp"
  alt="上下文菜单中的“编辑”和“删除”选项上方有一个预览内容卡片">

```tsx
import { View, Text as RNText } from 'react-native';
import {
  Host,
  ContextMenu,
  Button,
  RNHostView,
  Text,
} from '@expo/ui/swift-ui';

export default function ContextMenuWithPreviewExample() {
  return (
    <Host style={{ flex: 1 }}>
      <ContextMenu>
        <ContextMenu.Items>
          <Button
            label="Edit"
            onPress={() => console.log('Edit')}
          />
          <Button
            label="Delete"
            role="destructive"
            onPress={() => console.log('Delete')}
          />
        </ContextMenu.Items>
        <ContextMenu.Trigger>
          <Text>长按我</Text>
        </ContextMenu.Trigger>
        <ContextMenu.Preview>
          <RNHostView matchContents>
            <View
              style={{
                width: 200,
                height: 100,
                backgroundColor: '#f0f0f0',
                padding: 16,
              }}>
              <RNText>Preview content</RNText>
            </View>
          </RNHostView>
        </ContextMenu.Preview>
      </ContextMenu>
    </Host>
  );
}
```

</ComponentExample>

### 带有选择器的上下文菜单

<ComponentExample
  title="ContextMenuWithPickerExample.tsx"
  src="/static/images/expo-ui/examples/contextmenu-with-picker-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/contextmenu-with-picker-ios-dark.webp"
  alt="上下文菜单列出“操作”和一个带有子菜单箭头的“大小”行">

```tsx
import { useState } from 'react';
import {
  Host,
  ContextMenu,
  Button,
  Text,
  Picker,
} from '@expo/ui/swift-ui';
import { pickerStyle, tag } from '@expo/ui/swift-ui/modifiers';

export default function ContextMenuWithPickerExample() {
  const [selectedIndex, setSelectedIndex] = useState<
    number | undefined
  >(0);

  return (
    <Host style={{ flex: 1 }}>
      <ContextMenu>
        <ContextMenu.Items>
          <Button
            label="Action"
            onPress={() => console.log('Action')}
          />
          <Picker
            label="大小"
            modifiers={[pickerStyle('menu')]}
            selection={selectedIndex}
            onSelectionChange={setSelectedIndex}>
            {['小', '中', '大'].map((option, index) => (
              <Text key={index} modifiers={[tag(index)]}>
                {option}
              </Text>
            ))}
          </Picker>
        </ContextMenu.Items>
        <ContextMenu.Trigger>
          <Text>长按我</Text>
        </ContextMenu.Trigger>
      </ContextMenu>
    </Host>
  );
}
```

</ComponentExample>

### 禁用项

在菜单 `Button` 上使用 [`disabled(true)`](modifiers/#disableddisabled) 修饰符，可将其显示为灰色且不可交互。

<ComponentExample
  title="DisabledContextMenuItemExample.tsx"
  src="/static/images/expo-ui/examples/contextmenu-disabled-item-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/contextmenu-disabled-item-ios-dark.webp"
  alt="上下文菜单列出“编辑”和一个带锁图标的灰色“已锁定”项">

```tsx
import { Host, ContextMenu, Button, Text } from '@expo/ui/swift-ui';
import { disabled } from '@expo/ui/swift-ui/modifiers';

export default function DisabledContextMenuItemExample() {
  return (
    <Host style={{ flex: 1 }}>
      <ContextMenu>
        <ContextMenu.Items>
          <Button
            label="Edit"
            onPress={() => console.log('Edit')}
          />
          <Button
            label="已锁定"
            systemImage="lock"
            modifiers={[disabled(true)]}
            onPress={() => console.log('This never fires')}
          />
        </ContextMenu.Items>
        <ContextMenu.Trigger>
          <Text>长按我</Text>
        </ContextMenu.Trigger>
      </ContextMenu>
    </Host>
  );
}
```

</ComponentExample>

### 可选项（勾选标记）

`ContextMenu.Items` 中的 SwiftUI [`Toggle`](toggle) 在 `isOn` 为 `true` 时，会自动渲染为左侧带 SF Symbol、右侧带勾选标记的一行。

<ComponentExample
  title="CheckmarkContextMenuItemExample.tsx"
  src="/static/images/expo-ui/examples/contextmenu-checkmark-item-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/contextmenu-checkmark-item-ios-dark.webp"
  alt="上下文菜单中标有“固定”的一行，带有图钉图标和左侧的勾选标记">

```tsx
import { Host, ContextMenu, Toggle, Text } from '@expo/ui/swift-ui';
import { useState } from 'react';

export default function CheckmarkContextMenuItemExample() {
  const [pinned, setPinned] = useState(false);

  return (
    <Host style={{ flex: 1 }}>
      <ContextMenu>
        <ContextMenu.Items>
          <Toggle
            isOn={pinned}
            label="Pin"
            systemImage="pin"
            onIsOnChange={setPinned}
          />
        </ContextMenu.Items>
        <ContextMenu.Trigger>
          <Text>长按我</Text>
        </ContextMenu.Trigger>
      </ContextMenu>
    </Host>
  );
}
```

</ComponentExample>
