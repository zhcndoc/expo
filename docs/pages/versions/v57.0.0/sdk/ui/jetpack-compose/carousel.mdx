Expo UI 提供了三个轮播组件，分别对应官方 Jetpack Compose [Carousel](https://developer.android.com/develop/ui/compose/components/carousel) API：`HorizontalCenteredHeroCarousel`、`HorizontalMultiBrowseCarousel` 和 `HorizontalUncontainedCarousel`。

> **注意：** Carousel 是一个可水平滚动的组件，因此父级 `Host` 必须在滚动轴上提供有限宽度。请将 `matchContents={{ vertical: true }}` 与 `style={{ width: '100%' }}` 一起使用（或任何有限宽度）。有关详细信息，请参阅 [Host 参考中的匹配内容](host/#match-contents)。

## 用法

### HorizontalCenteredHeroCarousel

将一个较大的主视觉项居中放置在两个较小的预览项之间——非常适合展示电影海报等内容。

<ComponentExample
  title="CenteredHeroExample.tsx"
  src="/static/images/expo-ui/examples/carousel-centered-hero-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/carousel-centered-hero-android-dark.webp"
  alt="一个宽大的紫色主视觉幻灯片，两侧各有一个较窄的预览幻灯片">

```tsx
import {
  Host,
  HorizontalCenteredHeroCarousel,
  Box,
  Text,
} from '@expo/ui/jetpack-compose';
import {
  size,
  background,
  maskClip,
  Shapes,
} from '@expo/ui/jetpack-compose/modifiers';

export default function CenteredHeroExample() {
  const colors = [
    '#6200EE',
    '#03DAC5',
    '#FF5722',
    '#4CAF50',
    '#2196F3',
  ];

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <HorizontalCenteredHeroCarousel itemSpacing={8}>
        {colors.map((color, index) => (
          <Box
            key={index}
            contentAlignment="center"
            modifiers={[
              size(300, 200),
              maskClip(Shapes.RoundedCorner(28)),
              background(color),
            ]}>
            <Text color="#FFFFFF">Slide {index + 1}</Text>
          </Box>
        ))}
      </HorizontalCenteredHeroCarousel>
    </Host>
  );
}
```

</ComponentExample>

### HorizontalMultiBrowseCarousel

显示一个大项以及较小的预览项，让用户浏览接下来会出现的内容。

<ComponentExample
  title="MultiBrowseExample.tsx"
  src="/static/images/expo-ui/examples/carousel-multi-browse-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/carousel-multi-browse-android-dark.webp"
  alt="第一张卡片较大，后续预览卡片逐渐变窄">

```tsx
import {
  Host,
  HorizontalMultiBrowseCarousel,
  Box,
  Text,
} from '@expo/ui/jetpack-compose';
import {
  size,
  background,
} from '@expo/ui/jetpack-compose/modifiers';

export default function MultiBrowseExample() {
  const colors = [
    '#6200EE',
    '#03DAC5',
    '#FF5722',
    '#4CAF50',
    '#2196F3',
  ];

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <HorizontalMultiBrowseCarousel
        preferredItemWidth={200}
        itemSpacing={8}
        flingBehavior="singleAdvance">
        {colors.map((color, index) => (
          <Box
            key={index}
            contentAlignment="center"
            modifiers={[size(200, 180), background(color)]}>
            <Text color="#FFFFFF">卡片 {index + 1}</Text>
          </Box>
        ))}
      </HorizontalMultiBrowseCarousel>
    </Host>
  );
}
```

</ComponentExample>

### HorizontalUncontainedCarousel

每个项都有固定宽度，并支持自由形式滚动。

<ComponentExample
  title="UncontainedExample.tsx"
  src="/static/images/expo-ui/examples/carousel-uncontained-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/carousel-uncontained-android-dark.webp"
  alt="等宽照片卡片滚动到屏幕边缘之外">

```tsx
import {
  Host,
  HorizontalUncontainedCarousel,
  Box,
  Text,
} from '@expo/ui/jetpack-compose';
import {
  size,
  background,
} from '@expo/ui/jetpack-compose/modifiers';

export default function UncontainedExample() {
  const items = [
    'Photo 1',
    'Photo 2',
    'Photo 3',
    'Photo 4',
    'Photo 5',
  ];

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <HorizontalUncontainedCarousel
        itemWidth={160}
        itemSpacing={12}
        contentPadding={{ start: 16, top: 0, end: 16, bottom: 0 }}>
        {items.map(item => (
          <Box
            key={item}
            contentAlignment="center"
            modifiers={[size(160, 180), background('#3F51B5')]}>
            <Text color="#FFFFFF">{item}</Text>
          </Box>
        ))}
      </HorizontalUncontainedCarousel>
    </Host>
  );
}
```

</ComponentExample>
