```tsx
import { useColorScheme } from 'react-native';
import { Host, Text } from '@expo/ui';

export default function TextExample() {
  const colorScheme = useColorScheme();

  return (
    <Host matchContents>
      <Text textStyle={{ color: colorScheme === 'dark' ? '#FFFFFF' : '#000000' }}>
        Hello, world!
      </Text>
    </Host>
  );
}
```

</ComponentExample>

### 样式化文本

对排版相关属性（字体大小、字重、对齐方式）使用 [`textStyle`](#textstyle)。

<ComponentExample
  title="StyledTextExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-text-styled-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-text-styled-android-dark.webp',
    alt: '使用大号粗体显示的单词 Headline',
  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-text-styled-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-text-styled-ios-dark.webp',
    alt: '使用大号粗体显示的单词 Headline',
  }}>

```tsx
import { useColorScheme } from 'react-native';
import { Host, Text } from '@expo/ui';

export default function StyledTextExample() {
  const colorScheme = useColorScheme();

  return (
    <Host matchContents>
      <Text
        textStyle={{
          color: colorScheme === 'dark' ? '#FFFFFF' : '#000000',
          fontSize: 24,
          fontWeight: '700',
          textAlign: 'center',
        }}>
        Headline
      </Text>
    </Host>
  );
}
```

</ComponentExample>

### 截断长文本

使用 [`numberOfLines`](#numberoflines) 来限制长文本，并在末尾显示省略号。

<ComponentExample
  title="TruncatedTextExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-text-truncated-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-text-truncated-android-dark.webp',
    alt: '在屏幕边缘被截断的一行长文本',
  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-text-truncated-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-text-truncated-ios-dark.webp',
    alt: '使用省略号截断的一行长文本',
  }}>

```tsx
import { useColorScheme } from 'react-native';
import { Host, Text } from '@expo/ui';

export default function TruncatedTextExample() {
  const colorScheme = useColorScheme();

  return (
    <Host matchContents={{ vertical: true }} style={{ width: '100%' }}>
      <Text numberOfLines={1} textStyle={{ color: colorScheme === 'dark' ? '#FFFFFF' : '#000000' }}>
        A very long line of text that will be truncated when it does not fit on a single line.
      </Text>
    </Host>
  );
}
```

</ComponentExample>
