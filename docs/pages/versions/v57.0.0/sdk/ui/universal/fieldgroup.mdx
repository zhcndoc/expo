一个可滚动的容器，用于容纳分组的设置样式行，外观类似 iOS 设置界面。将 `FieldGroup.Section`（用于显式分组）、`FieldGroup.SectionHeader` 和 `FieldGroup.SectionFooter` 插槽放置其中。

### 分组表单

<ComponentExample
  title="FieldGroupExample.tsx"
  android={{

    darkSrc: '/static/images/expo-ui/examples/universal-fieldgroup-basic-android-dark.webp',

  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-fieldgroup-basic-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-fieldgroup-basic-ios-dark.webp',
    alt: '一个「通知」分区，其中「推送」已开启、「电子邮件」已关闭，其下方是一个「About」分区',
  }}>

```tsx
import { useState } from 'react';
import { useColorScheme } from 'react-native';
import { Host, FieldGroup, Switch, Text } from '@expo/ui';

export default function FieldGroupExample() {
  const [notifications, setNotifications] = useState(true);
  const [analytics, setAnalytics] = useState(false);
  const colorScheme = useColorScheme();
  const ink = { color: colorScheme === 'dark' ? '#FFFFFF' : '#000000' };

  return (
    <Host style={{ flex: 1 }}>
      <FieldGroup>
        <FieldGroup.Section title="通知">
          <Switch label="推送" value={notifications} onValueChange={setNotifications} />
          <Switch label="电子邮件" value={analytics} onValueChange={setAnalytics} />
        </FieldGroup.Section>

        <FieldGroup.Section title="About">
          <Text textStyle={ink}>Version 1.0.0</Text>
        </FieldGroup.Section>
      </FieldGroup>
    </Host>
  );
}
```

</ComponentExample>

### 自定义分区标题和页脚

使用 `FieldGroup.SectionHeader` 和 `FieldGroup.SectionFooter`，可以在默认 `title` 文本的位置渲染完全样式化的标题／页脚插槽。

<ComponentExample
  title="FieldGroupSlotsExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-fieldgroup-slots-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-fieldgroup-slots-android-dark.webp',
    alt: '一个「隐私」分区，包含一个「共享使用情况」开关及其下方的脚注',
  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-fieldgroup-slots-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-fieldgroup-slots-ios-dark.webp',
    alt: '一个「隐私」分区，包含一个「共享使用情况」开关及其下方的脚注',
  }}>

```tsx
import { useState } from 'react';
import { useColorScheme } from 'react-native';
import { Host, FieldGroup, Switch, Text } from '@expo/ui';

export default function FieldGroupSlotsExample() {
  const [enabled, setEnabled] = useState(false);
  const colorScheme = useColorScheme();
  const ink = { color: colorScheme === 'dark' ? '#FFFFFF' : '#000000' };

  return (
    <Host style={{ flex: 1 }}>
      <FieldGroup>
        <FieldGroup.Section>
          <FieldGroup.SectionHeader>
            <Text textStyle={{ ...ink, fontSize: 16, fontWeight: '700' }}>Privacy</Text>
          </FieldGroup.SectionHeader>

          <Switch label="共享使用情况" value={enabled} onValueChange={setEnabled} />

          <FieldGroup.SectionFooter>
            <Text textStyle={{ fontSize: 12, color: '#8E8E93' }}>
              帮助我们改进应用。你可以随时关闭此功能。
            </Text>
          </FieldGroup.SectionFooter>
        </FieldGroup.Section>
      </FieldGroup>
    </Host>
  );
}
```

</ComponentExample>
