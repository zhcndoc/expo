Expo UI Card 与官方 Jetpack Compose [Card API](https://developer.android.com/develop/ui/compose/components/card) 相匹配，可在带样式的表面容器中显示内容，并支持可选的阴影和描边。`Card` 组件会渲染[填充卡片](https://developer.android.com/develop/ui/compose/components/card#filled)，而 `ElevatedCard` 和 `OutlinedCard` 分别提供凸起和带边框的变体。

## 用法

### 基本卡片

<ComponentExample
  title="BasicCardExample.tsx"
  src="/static/images/expo-ui/examples/card-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/card-basic-android-dark.webp"
  alt="一张填充式 Material 3 卡片，其中包含一行带内边距的文本">

```tsx
import { Host, Card, Text } from '@expo/ui/jetpack-compose';
import { paddingAll } from '@expo/ui/jetpack-compose/modifiers';

export default function BasicCardExample() {
  return (
    <Host matchContents>
      <Card>
        <Text modifiers={[paddingAll(16)]}>
          This is a basic card with default styling.
        </Text>
      </Card>
    </Host>
  );
}
```

</ComponentExample>

### 卡片类型

使用 `Card`（填充式）、`ElevatedCard` 或 `OutlinedCard` 来应用不同的样式。

<ComponentExample
  title="CardTypesExample.tsx"
  src="/static/images/expo-ui/examples/card-types-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/card-types-android-dark.webp"
  alt="按列堆叠的填充式、凸起式和带描边的卡片">

```tsx
import {
  Host,
  Card,
  ElevatedCard,
  OutlinedCard,
  Text,
  Column,
} from '@expo/ui/jetpack-compose';
import { paddingAll } from '@expo/ui/jetpack-compose/modifiers';

export default function CardTypesExample() {
  return (
    <Host matchContents>
      <Column verticalArrangement={{ spacedBy: 12 }}>
        <Card>
          <Text modifiers={[paddingAll(16)]}>Filled card</Text>
        </Card>
        <ElevatedCard>
          <Text modifiers={[paddingAll(16)]}>Elevated card</Text>
        </ElevatedCard>
        <OutlinedCard>
          <Text modifiers={[paddingAll(16)]}>Outlined card</Text>
        </OutlinedCard>
      </Column>
    </Host>
  );
}
```

</ComponentExample>

### 自定义阴影

使用 `elevation` 属性（单位为 dp）控制阴影深度。阴影高度对 `ElevatedCard` 的效果最明显，因为它使用阴影高度。填充式 `Card` 默认使用色调阴影高度，因此调整效果可能不明显。

<ComponentExample
  title="ElevatedCardExample.tsx"
  src="/static/images/expo-ui/examples/card-elevated-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/card-elevated-android-dark.webp"
  alt="一张阴影高度为 8dp、带有投影的凸起式卡片">

```tsx
import { Host, ElevatedCard, Text } from '@expo/ui/jetpack-compose';
import { paddingAll } from '@expo/ui/jetpack-compose/modifiers';

export default function ElevatedCardExample() {
  return (
    <Host matchContents>
      <ElevatedCard elevation={8}>
        <Text modifiers={[paddingAll(16)]}>
          Card with 8dp elevation
        </Text>
      </ElevatedCard>
    </Host>
  );
}
```

</ComponentExample>

### 自定义边框

`Card` 和 `OutlinedCard` 接受 `border` 属性，可用于自定义描边宽度和颜色。

<ComponentExample
  title="OutlinedCardExample.tsx"
  src="/static/images/expo-ui/examples/card-outlined-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/card-outlined-android-dark.webp"
  alt="一张带有 2dp 紫色边框的带描边卡片">

```tsx
import { Host, OutlinedCard, Text } from '@expo/ui/jetpack-compose';
import { paddingAll } from '@expo/ui/jetpack-compose/modifiers';

export default function OutlinedCardExample() {
  return (
    <Host matchContents>
      <OutlinedCard border={{ width: 2, color: '#6200EE' }}>
        <Text modifiers={[paddingAll(16)]}>
          Card with custom purple border
        </Text>
      </OutlinedCard>
    </Host>
  );
}
```

</ComponentExample>
