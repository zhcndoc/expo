## 用法

`Row` 可水平排列子组件。使用 `horizontalArrangement` 和 `verticalAlignment` 控制间距和对齐方式。

<ComponentExample
  title="RowExample.tsx"
  src="/static/images/expo-ui/examples/row-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/row-basic-android-dark.webp"
  alt="三个标签在一行中均匀分布">

```tsx
import {
  Host,
  Row,
  Text,
  useMaterialColors,
} from '@expo/ui/jetpack-compose';
import {
  fillMaxWidth,
  height,
} from '@expo/ui/jetpack-compose/modifiers';

export default function RowExample() {
  const colors = useMaterialColors();

  return (
    <Host style={{ flex: 1 }}>
      <Row
        horizontalArrangement="spaceEvenly"
        verticalAlignment="center"
        modifiers={[fillMaxWidth(), height(60)]}>
        <Text style={{ fontSize: 28 }} color={colors.onBackground}>
          Item 1
        </Text>
        <Text style={{ fontSize: 28 }} color={colors.onBackground}>
          Item 2
        </Text>
        <Text style={{ fontSize: 28 }} color={colors.onBackground}>
          Item 3
        </Text>
      </Row>
    </Host>
  );
}
```

</ComponentExample>
