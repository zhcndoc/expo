## 用法

### 切换开关

<ComponentExample
  title="ToggleSwitchExample.tsx"
  src="/static/images/expo-ui/examples/switch-toggle-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/switch-toggle-android-dark.webp"
  alt="一个未选中的 Material 3 开关">

```tsx
import { useState } from 'react';
import { Host, Switch } from '@expo/ui/jetpack-compose';

export default function ToggleSwitchExample() {
  const [checked, setChecked] = useState(false);

  return (
    <Host matchContents>
      <Switch value={checked} onCheckedChange={setChecked} />
    </Host>
  );
}
```

</ComponentExample>

### 自定义颜色

<ComponentExample
  title="CustomColorsExample.tsx"
  src="/static/images/expo-ui/examples/switch-colors-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/switch-colors-android-dark.webp"
  alt="一个已选中的开关，其紫色滑块位于淡紫色轨道上">

```tsx
import { useState } from 'react';
import { Host, Switch } from '@expo/ui/jetpack-compose';

export default function CustomColorsExample() {
  const [checked, setChecked] = useState(false);

  return (
    <Host matchContents>
      <Switch
        value={checked}
        onCheckedChange={setChecked}
        colors={{
          checkedThumbColor: '#6200EE',
          checkedTrackColor: '#EDE9FE',
          uncheckedThumbColor: '#9CA3AF',
          uncheckedTrackColor: '#F3F4F6',
          uncheckedBorderColor: '#D1D5DB',
        }}
      />
    </Host>
  );
}
```

</ComponentExample>

### 自定义滑块内容

使用 `Switch.ThumbContent` 在滑块内渲染自定义元素。`Switch.DefaultIconSize` 可提供 M3 默认图标尺寸，让内容完美适配。

<ComponentExample
  title="ThumbContentExample.tsx"
  src="/static/images/expo-ui/examples/switch-thumb-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/switch-thumb-android-dark.webp"
  alt="一个已选中的开关，其紫色滑块内含一个白色圆形，呈现出圆环效果">

```tsx
import { useState } from 'react';
import { Host, Switch, Box } from '@expo/ui/jetpack-compose';
import {
  size,
  clip,
  background,
  Shapes,
} from '@expo/ui/jetpack-compose/modifiers';

export default function ThumbContentExample() {
  const [checked, setChecked] = useState(false);

  return (
    <Host matchContents>
      <Switch
        value={checked}
        onCheckedChange={setChecked}
        colors={{
          checkedThumbColor: '#7C3AED',
          checkedTrackColor: '#EDE9FE',
          checkedIconColor: '#7C3AED',
          uncheckedThumbColor: '#9CA3AF',
          uncheckedTrackColor: '#F3F4F6',
          uncheckedBorderColor: '#D1D5DB',
          uncheckedIconColor: '#9CA3AF',
        }}>
        <Switch.ThumbContent>
          <Box
            modifiers={[
              size(Switch.DefaultIconSize, Switch.DefaultIconSize),
              clip(Shapes.Circle),
              background(checked ? '#FFFFFF' : '#E5E7EB'),
            ]}
          />
        </Switch.ThumbContent>
      </Switch>
    </Host>
  );
}
```

</ComponentExample>
