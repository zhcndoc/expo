Expo UI AlertDialog 与官方 Jetpack Compose [AlertDialog](https://developer.android.com/develop/ui/compose/components/dialog) API 相匹配。内容通过插槽子组件（`AlertDialog.Title`、`AlertDialog.Text`、`AlertDialog.ConfirmButton`、`AlertDialog.DismissButton`、`AlertDialog.Icon`）提供，这些组件直接映射到 Compose 的插槽参数。

## 用法

### 基本警告对话框

<ComponentExample
  title="BasicAlertDialogExample.tsx"
  src="/static/images/expo-ui/examples/alertdialog-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/alertdialog-basic-android-dark.webp"
  alt="一个 Material 3 警告对话框，标题为确认操作，包含辅助文本以及取消和确认按钮">

```tsx
import { useState } from 'react';
import {
  Host,
  AlertDialog,
  Button,
  TextButton,
  Text,
} from '@expo/ui/jetpack-compose';

export default function BasicAlertDialogExample() {
  const [visible, setVisible] = useState(false);

  return (
    <Host matchContents>
      <Button onClick={() => setVisible(true)}>
        <Text>Show alert</Text>
      </Button>
      {visible && (
        <AlertDialog onDismissRequest={() => setVisible(false)}>
          <AlertDialog.Title>
            <Text>Confirm action</Text>
          </AlertDialog.Title>
          <AlertDialog.Text>
            <Text>Are you sure you want to proceed?</Text>
          </AlertDialog.Text>
          <AlertDialog.ConfirmButton>
            <TextButton onClick={() => setVisible(false)}>
              <Text>Confirm</Text>
            </TextButton>
          </AlertDialog.ConfirmButton>
          <AlertDialog.DismissButton>
            <TextButton onClick={() => setVisible(false)}>
              <Text>Cancel</Text>
            </TextButton>
          </AlertDialog.DismissButton>
        </AlertDialog>
      )}
    </Host>
  );
}
```

</ComponentExample>

### 自定义颜色

<ComponentExample
  title="CustomColorsExample.tsx"
  src="/static/images/expo-ui/examples/alertdialog-custom-colors-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/alertdialog-custom-colors-android-dark.webp"
  alt="一个深海军蓝容器、浅色文本的警告对话框，覆盖了主题颜色">

```tsx
import { useState } from 'react';
import {
  Host,
  AlertDialog,
  Button,
  TextButton,
  Text,
} from '@expo/ui/jetpack-compose';

export default function CustomColorsExample() {
  const [visible, setVisible] = useState(false);

  return (
    <Host matchContents>
      <Button onClick={() => setVisible(true)}>
        <Text>Show alert</Text>
      </Button>
      {visible && (
        <AlertDialog
          onDismissRequest={() => setVisible(false)}
          colors={{
            containerColor: '#1E1E2E',
            titleContentColor: '#CDD6F4',
            textContentColor: '#BAC2DE',
          }}>
          <AlertDialog.Title>
            <Text>Custom Dialog</Text>
          </AlertDialog.Title>
          <AlertDialog.Text>
            <Text>This dialog uses custom colors.</Text>
          </AlertDialog.Text>
          <AlertDialog.ConfirmButton>
            <TextButton onClick={() => setVisible(false)}>
              <Text>OK</Text>
            </TextButton>
          </AlertDialog.ConfirmButton>
          <AlertDialog.DismissButton>
            <TextButton onClick={() => setVisible(false)}>
              <Text>Cancel</Text>
            </TextButton>
          </AlertDialog.DismissButton>
        </AlertDialog>
      )}
    </Host>
  );
}
```

</ComponentExample>

### 带图标

<ComponentExample
  title="IconDialogExample.tsx"
  src="/static/images/expo-ui/examples/alertdialog-icon-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/alertdialog-icon-android-dark.webp"
  alt="一个带有信息图标的警告对话框，图标居中显示在标题“带图标的对话框”上方">

```tsx
import { useState } from 'react';
import {
  Host,
  AlertDialog,
  Button,
  TextButton,
  Text,
  Icon,
} from '@expo/ui/jetpack-compose';

export default function IconDialogExample() {
  const [visible, setVisible] = useState(false);

  return (
    <Host matchContents>
      <Button onClick={() => setVisible(true)}>
        <Text>Show alert</Text>
      </Button>
      {visible && (
        <AlertDialog onDismissRequest={() => setVisible(false)}>
          <AlertDialog.Icon>
            {/* Replace with your own icon asset */}
            <Icon source={require('./info-icon.xml')} />
          </AlertDialog.Icon>
          <AlertDialog.Title>
            <Text>Dialog with Icon</Text>
          </AlertDialog.Title>
          <AlertDialog.Text>
            <Text>This dialog has an icon above the title.</Text>
          </AlertDialog.Text>
          <AlertDialog.ConfirmButton>
            <TextButton onClick={() => setVisible(false)}>
              <Text>OK</Text>
            </TextButton>
          </AlertDialog.ConfirmButton>
        </AlertDialog>
      )}
    </Host>
  );
}
```

</ComponentExample>
