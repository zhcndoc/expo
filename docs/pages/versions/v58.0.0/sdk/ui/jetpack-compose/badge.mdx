Expo UI Badge 与官方 Jetpack Compose [`Badge`](https://developer.android.com/develop/ui/compose/components/badges) API 相匹配。它会渲染为一个小型彩色指示点，也可以显示计数等内容。

## 用法

### 指示点

没有子元素的徽章会渲染为一个小型圆点指示器。

<ComponentExample
  title="BadgeDot.tsx"
  src="/static/images/expo-ui/examples/badge-dot-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/badge-dot-android-dark.webp"
  alt="一个小型红色 Material 3 徽章圆点，位于其他内容为空的屏幕上">

```tsx
import { Host, Badge } from '@expo/ui/jetpack-compose';

export default function BadgeDot() {
  return (
    <Host matchContents>
      <Badge />
    </Host>
  );
}
```

</ComponentExample>

### 带计数的徽章

传入一个 `Text` 子元素来显示数字或标签。

<ComponentExample
  title="BadgeCount.tsx"
  src="/static/images/expo-ui/examples/badge-count-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/badge-count-android-dark.webp"
  alt="一个红色圆形徽章，以白色显示计数 3">

```tsx
import { Host, Badge, Text } from '@expo/ui/jetpack-compose';

export default function BadgeCount() {
  return (
    <Host matchContents>
      <Badge containerColor="#EF5350" contentColor="#FFFFFF">
        <Text>3</Text>
      </Badge>
    </Host>
  );
}
```

</ComponentExample>
