Expo UI Shape 与官方 Jetpack Compose [Shapes](https://developer.android.com/develop/ui/compose/graphics/draw/shapes) API 相匹配，并提供一组子组件，用于绘制星形、圆形、矩形、药丸形和多边形等几何形状。

## 用法

### 基本形状

使用 `Shape` 子组件渲染常见形状。

<ComponentExample
  title="BasicShapesExample.tsx"
  src="/static/images/expo-ui/examples/shape-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/shape-basic-android-dark.webp"
  alt="一排金色星形、蓝色圆形、绿色正方形和红色药丸形">

```tsx
import { Host, Shape, Row } from '@expo/ui/jetpack-compose';
import { size } from '@expo/ui/jetpack-compose/modifiers';

export default function BasicShapesExample() {
  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <Row
        horizontalArrangement={{ spacedBy: 16 }}
        verticalAlignment="center">
        <Shape.Star
          radius={1}
          innerRadius={0.5}
          color="#FFD700"
          modifiers={[size(80, 80)]}
        />
        <Shape.Circle
          radius={1}
          color="#4285F4"
          modifiers={[size(80, 80)]}
        />
        <Shape.Rectangle
          color="#34A853"
          modifiers={[size(80, 80)]}
        />
        <Shape.Pill color="#EA4335" modifiers={[size(100, 50)]} />
      </Row>
    </Host>
  );
}
```

</ComponentExample>

### 圆角形状

使用 `cornerRounding` 和 `smoothing` 自定义形状的外观。

<ComponentExample
  title="RoundedShapesExample.tsx"
  src="/static/images/expo-ui/examples/shape-rounded-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/shape-rounded-android-dark.webp"
  alt="一个紫色圆角矩形，旁边是一个只有顶部带圆角的橙色矩形">

```tsx
import { Host, Shape, Row } from '@expo/ui/jetpack-compose';
import { size } from '@expo/ui/jetpack-compose/modifiers';

export default function RoundedShapesExample() {
  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <Row
        horizontalArrangement={{ spacedBy: 16 }}
        verticalAlignment="center">
        <Shape.Rectangle
          cornerRounding={0.2}
          smoothing={0.5}
          color="#9C27B0"
          modifiers={[size(100, 80)]}
        />
        <Shape.RoundedCorner
          cornerRadii={{
            topStart: 20,
            topEnd: 20,
            bottomStart: 0,
            bottomEnd: 0,
          }}
          color="#FF5722"
          modifiers={[size(100, 80)]}
        />
      </Row>
    </Host>
  );
}
```

</ComponentExample>

### 多边形和星形变体

使用 `verticesCount` 和 `innerRadius` 控制形状的几何结构。

<ComponentExample
  title="PolygonShapesExample.tsx"
  src="/static/images/expo-ui/examples/shape-polygon-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/shape-polygon-android-dark.webp"
  alt="一个青色六边形、一个橙色八角星，以及一个粉色六角药丸星">

```tsx
import { Host, Shape, Row } from '@expo/ui/jetpack-compose';
import { size } from '@expo/ui/jetpack-compose/modifiers';

export default function PolygonShapesExample() {
  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <Row
        horizontalArrangement={{ spacedBy: 16 }}
        verticalAlignment="center">
        <Shape.Polygon
          verticesCount={6}
          cornerRounding={0.05}
          color="#00BCD4"
          modifiers={[size(80, 80)]}
        />
        <Shape.Star
          verticesCount={8}
          radius={1}
          innerRadius={0.4}
          cornerRounding={0.025}
          color="#FF9800"
          modifiers={[size(80, 80)]}
        />
        <Shape.PillStar
          verticesCount={6}
          innerRadius={0.5}
          color="#E91E63"
          modifiers={[size(80, 80)]}
        />
      </Row>
    </Host>
  );
}
```

</ComponentExample>
