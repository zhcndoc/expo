Expo UI 提供了三个与官方 Jetpack Compose [TextField API](https://developer.android.com/develop/ui/compose/text/user-input) 对应的文本字段组件：`TextField`（填充式）、`OutlinedTextField`（轮廓式边框）和 `BasicTextField`（无样式）。Material 变体 `TextField` 和 `OutlinedTextField` 共享相同的 props，并支持使用可组合插槽子元素设置标签、占位符、图标、前缀、后缀和辅助文本。`BasicTextField` 没有 Material 装饰，因此需要自行提供装饰。

| 类型   | 外观                                       | 用途                                                                    |
| ------ | ------------------------------------------ | ----------------------------------------------------------------------- |
| 填充式 | 实色背景，带有底部指示线。                 | 遵循 Material3 设计的默认文本输入样式。适用于大多数表单和输入字段。     |
| 轮廓式 | 透明背景，带有轮廓边框。                   | 提供清晰视觉边界的替代样式。当填充式字段与背景融为一体时使用。          |
| 基础式 | 没有容器、指示线或内边距。只有可编辑文本。 | 完全自定义样式的输入框。自行设置样式，并通过 `DecorationBox` 添加装饰。 |

## 用法

### 非受控文本字段

将 [`useNativeState`](usenativestate) 可观察对象绑定到 `value`。字段会自行跟踪用户输入，并可通过 `text.value` 读取当前值。此处展示的填充式样式是默认的 Material3 文本输入样式。

<ComponentExample
  title="UncontrolledTextFieldExample.tsx"
  src="/static/images/expo-ui/examples/composetextfield-uncontrolled-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/composetextfield-uncontrolled-android-dark.webp"
  alt="一个带有 Username 标签的填充式 Material 3 文本字段">

```tsx
import {
  Host,
  TextField,
  Text,
  useNativeState,
} from '@expo/ui/jetpack-compose';

export default function UncontrolledTextFieldExample() {
  const text = useNativeState('');

  return (
    <Host matchContents>
      <TextField value={text}>
        <TextField.Label>
          <Text>Username</Text>
        </TextField.Label>
      </TextField>
    </Host>
  );
}
```

</ComponentExample>

### 受控文本字段

将 [`useNativeState`](usenativestate) 可观察对象作为 `value` 传入，并使用 `onValueChange` worklet，在写回输入内容前对其进行转换或验证。下面的示例会在输入时将文本转换为大写。

> **注意：** Worklet 需要安装 [`react-native-worklets`](https://docs.swmansion.com/react-native-worklets/)。

<ComponentExample
  title="ControlledTextFieldExample.tsx"
  src="/static/images/expo-ui/examples/composetextfield-controlled-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/composetextfield-controlled-android-dark.webp"
  alt="一个标有 Name 的文本字段，内容为输入时转换成大写的 ADA LOVELACE">

```tsx
import {
  Host,
  TextField,
  Text,
  useNativeState,
} from '@expo/ui/jetpack-compose';
import { useCallback } from 'react';

export default function ControlledTextFieldExample() {
  const text = useNativeState('');

  const handleValueChange = useCallback(
    (value: string) => {
      'worklet';
      text.value = value.toUpperCase();
    },
    [text]
  );

  return (
    <Host matchContents>
      <TextField value={text} onValueChange={handleValueChange}>
        <TextField.Label>
          <Text>Name</Text>
        </TextField.Label>
      </TextField>
    </Host>
  );
}
```

</ComponentExample>

### 轮廓式文本字段

使用 `OutlinedTextField` 创建带轮廓边框而非填充背景的文本字段。

<ComponentExample
  title="OutlinedTextFieldExample.tsx"

  darkSrc="/static/images/expo-ui/examples/composetextfield-outlined-android-dark.webp"
  alt="一个处于焦点状态的轮廓式文本字段，Email 标签位于边框凹口中，并显示占位符">

```tsx
import {
  Host,
  OutlinedTextField,
  Text,
  useNativeState,
} from '@expo/ui/jetpack-compose';

export default function OutlinedTextFieldExample() {
  const text = useNativeState('');

  return (
    <Host matchContents>
      <OutlinedTextField value={text}>
        <OutlinedTextField.Label>
          <Text>Email</Text>
        </OutlinedTextField.Label>

          <Text>you@example.com</Text>
        </OutlinedTextField.Placeholder>
      </OutlinedTextField>
    </Host>
  );
}
```

</ComponentExample>

### 基础文本字段

`BasicTextField` 是无样式的 Compose 基础组件，没有容器、指示线或内边距。使用[修饰符](modifiers)自行设置样式，并通过 `DecorationBox` 提供装饰，将 `InnerTextField` 放置在可编辑文本应显示的位置。将占位符内容包裹在 `Placeholder` 中，使其仅在字段为空时显示；显示状态会根据字段文本由原生端切换。

<ComponentExample
  title="BasicTextFieldExample.tsx"
  src="/static/images/expo-ui/examples/composetextfield-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/composetextfield-basic-android-dark.webp"
  alt="一个无样式文本字段，呈现为全宽圆角灰色胶囊形状，并带有 Search 占位符">

```tsx
import {
  Host,
  BasicTextField,
  Box,
  Text,
  useNativeState,
} from '@expo/ui/jetpack-compose';
import {
  background,
  clip,
  fillMaxWidth,
  padding,
  Shapes,
} from '@expo/ui/jetpack-compose/modifiers';

export default function BasicTextFieldExample() {
  const value = useNativeState('');

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <BasicTextField
        cursorColor="#7c3aed"
        value={value}
        modifiers={[
          fillMaxWidth(),
          clip(Shapes.RoundedCorner(12)),
          background('#f3f4f6'),
          padding(12, 10, 12, 10),
        ]}>
        <BasicTextField.DecorationBox>
          <Box>
            <BasicTextField.Placeholder>
              <Text color="#9ca3af">Search…</Text>
            </BasicTextField.Placeholder>
            <BasicTextField.InnerTextField />
          </Box>
        </BasicTextField.DecorationBox>
      </BasicTextField>
    </Host>
  );
}
```

</ComponentExample>

### 插槽

`TextField` 和 `OutlinedTextField` 均支持 7 个与 Compose API 对应的可组合插槽：`Label`、`Placeholder`、`LeadingIcon`、`TrailingIcon`、`Prefix`、`Suffix` 和 `SupportingText`。

<ComponentExample
  title="TextFieldSlotsExample.tsx"
  src="/static/images/expo-ui/examples/composetextfield-slots-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/composetextfield-slots-android-dark.webp"
  alt="一个处于焦点状态的文本字段，显示金钱图标、Price 标签、美元前缀、占位符、USD 后缀和辅助文本">

```tsx
import {
  Host,
  TextField,
  Text,
  useNativeState,
} from '@expo/ui/jetpack-compose';

export default function TextFieldSlotsExample() {
  const text = useNativeState('');

  return (
    <Host matchContents>
      <TextField value={text}>
        <TextField.Label>
          <Text>Price</Text>
        </TextField.Label>
        <TextField.Placeholder>
          <Text>0.00</Text>
        </TextField.Placeholder>
        <TextField.LeadingIcon>
          <Text>💰</Text>
        </TextField.LeadingIcon>
        <TextField.Prefix>
          <Text>$</Text>
        </TextField.Prefix>
        <TextField.Suffix>
          <Text>USD</Text>
        </TextField.Suffix>
        <TextField.SupportingText>
          <Text>Enter the amount</Text>
        </TextField.SupportingText>
      </TextField>
    </Host>
  );
}
```

</ComponentExample>

### 键盘选项

使用 `keyboardOptions` prop 配置键盘类型、大小写、自动更正和 IME 操作。

  alt="一个处于焦点状态的电子邮件字段，位于电子邮件键盘上方；键盘上有专用的 @ 键和完成操作键">

```tsx
import {
  Host,
  TextField,
  Text,
  useNativeState,
} from '@expo/ui/jetpack-compose';

export default function KeyboardOptionsExample() {
  const text = useNativeState('');

  return (
    <Host matchContents>
      <TextField
        value={text}
        singleLine
        keyboardOptions={{
          keyboardType: 'email',
          capitalization: 'none',
          autoCorrectEnabled: false,
          imeAction: 'done',

        <TextField.Label>
          <Text>Email</Text>
        </TextField.Label>
      </TextField>

}
```

### 键盘操作

使用 `keyboardActions` prop 处理 IME 操作按钮的按下事件。触发的回调取决于 `keyboardOptions` 中设置的 `imeAction`。每个回调都会接收当前文本值。

<ComponentExample
  title="KeyboardActionsExample.tsx"
  src="/static/images/expo-ui/examples/composetextfield-keyboard-actions-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/composetextfield-keyboard-actions-android-dark.webp"
  alt="一个处于焦点状态的搜索字段，位于键盘上方，键盘操作键是放大镜">

```tsx
import {
  Host,
  TextField,
  Text,
  useNativeState,
} from '@expo/ui/jetpack-compose';

export default function KeyboardActionsExample() {
  const text = useNativeState('');

  return (
    <Host matchContents>
      <TextField
        value={text}
        singleLine
        keyboardOptions={{ imeAction: 'search' }}
        keyboardActions={{
          onSearch: value => console.log('Searched:', value),
        }}>
        <TextField.Label>
          <Text>Search</Text>
        </TextField.Label>
      </TextField>
    </Host>
  );
}
```

</ComponentExample>

### 命令式 ref

使用 ref 命令式地设置文本、清空字段、更改选区或移动焦点。

<ComponentExample
  title="ImperativeRefExample.tsx"
  src="/static/images/expo-ui/examples/composetextfield-imperative-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/composetextfield-imperative-android-dark.webp"
  alt="一个文本字段，下方有五个按钮：Set text、Clear、Select first word、Focus 和 Blur">

```tsx
import { useRef } from 'react';
import {
  Host,
  TextField,
  TextFieldRef,
  Button,
  Row,
  Text,
  Column,
  useNativeState,
} from '@expo/ui/jetpack-compose';
import { padding } from '@expo/ui/jetpack-compose/modifiers';

export default function ImperativeRefExample() {
  const ref = useRef<TextFieldRef>(null);
  const text = useNativeState('');

  return (
    <Host matchContents>
      <Column>
        <TextField ref={ref} value={text} singleLine>
          <TextField.Label>
            <Text>Name</Text>
          </TextField.Label>
        </TextField>
        <Row
          horizontalArrangement={{ spacedBy: 8 }}
          modifiers={[padding(8, 0, 0, 0)]}>
          <Button
            onClick={() => ref.current?.setText('Hello world')}>
            <Text>Set text</Text>
          </Button>
          <Button onClick={() => ref.current?.clear()}>
            <Text>Clear</Text>
          </Button>
          <Button onClick={() => ref.current?.setSelection(0, 5)}>
            <Text>Select first word</Text>
          </Button>
        </Row>
        <Row
          horizontalArrangement={{ spacedBy: 8 }}
          modifiers={[padding(8, 0, 0, 0)]}>
          <Button onClick={() => ref.current?.focus()}>
            <Text>Focus</Text>
          </Button>
          <Button onClick={() => ref.current?.blur()}>
            <Text>Blur</Text>
          </Button>
        </Row>
      </Column>
    </Host>
  );
}
```

</ComponentExample>

### Worklet 文本遮罩

当 `onValueChange` 使用 `'worklet'` 指令标记时，它会在 UI 线程上同步运行，因此回调中对 [`useNativeState`](usenativestate) 可观察对象的写入会在下一帧前生效。输入文本与遮罩后的文本之间不会闪烁。下面的示例会在用户输入时对电话号码进行遮罩，并从 worklet 同时写入 `value` 和 `selection`，使光标保持在格式化后文本的末尾。

> **注意：** Worklet 需要安装 [`react-native-worklets`](https://docs.swmansion.com/react-native-worklets/)。

<ComponentExample
  title="WorkletPhoneMaskExample.tsx"
  src="/static/images/expo-ui/examples/composetextfield-worklet-mask-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/composetextfield-worklet-mask-android-dark.webp"
  alt="一个全宽文本字段，将 2125559876 遮罩为格式化的电话号码">

```tsx
import {
  Host,
  TextField,
  Text,
  useNativeState,
} from '@expo/ui/jetpack-compose';
import { fillMaxWidth } from '@expo/ui/jetpack-compose/modifiers';
import { useCallback } from 'react';

export default function WorkletPhoneMaskExample() {
  const phone = useNativeState('');
  const selection = useNativeState({ start: 0, end: 0 });

  const handleValueChange = useCallback(
    (v: string) => {
      'worklet';
      const digits = v.replace(/\D/g, '').slice(0, 10);
      let formatted = digits;
      if (digits.length > 6) {
        formatted = `(${digits.slice(0, 3)}) ${digits.slice(3, 6)}-${digits.slice(6)}`;
      } else if (digits.length > 3) {
        formatted = `(${digits.slice(0, 3)}) ${digits.slice(3)}`;
      }
      if (formatted !== v) {
        phone.value = formatted;
        // Snaps to end for demo. Real masks need smarter cursor handling.
        selection.value = {
          start: formatted.length,
          end: formatted.length,
        };
      }
    },
    [phone, selection]
  );

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <TextField
        value={phone}
        selection={selection}
        keyboardOptions={{ keyboardType: 'phone' }}
        modifiers={[fillMaxWidth()]}
        onValueChange={handleValueChange}>
        <TextField.Placeholder>
          <Text>(555) 123-4567</Text>
        </TextField.Placeholder>
      </TextField>
    </Host>
  );
}
```

</ComponentExample>
