## 用法

### 基本滑块

<ComponentExample
  title="BasicSliderExample.tsx"
  src="/static/images/expo-ui/examples/slider-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/slider-basic-android-dark.webp"
  alt="一个滑块，滑块拇指位于中点">

```tsx
import { useState } from 'react';
import { Host, Slider } from '@expo/ui/jetpack-compose';

export default function BasicSliderExample() {
  const [value, setValue] = useState(0.5);

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <Slider value={value} onValueChange={setValue} />
    </Host>
  );
}
```

</ComponentExample>

### 自定义范围的滑块

使用 `min` 和 `max` 属性定义滑块的取值范围。

<ComponentExample
  title="CustomRangeSliderExample.tsx"
  src="/static/images/expo-ui/examples/slider-custom-range-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/slider-custom-range-android-dark.webp"
  alt="一个范围为 0 到 100、滑块拇指位于 50 的滑块">

```tsx
import { useState } from 'react';
import { Host, Slider } from '@expo/ui/jetpack-compose';

export default function CustomRangeSliderExample() {
  const [value, setValue] = useState(50);

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <Slider
        value={value}
        min={0}
        max={100}
        onValueChange={setValue}
      />
    </Host>
  );
}
```

</ComponentExample>

### 带步进的滑块

使用 `steps` 属性定义离散增量。将 `steps` 设为 `0` 可使用连续值。

<ComponentExample
  title="SteppedSliderExample.tsx"
  src="/static/images/expo-ui/examples/slider-stepped-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/slider-stepped-android-dark.webp"
  alt="一个带有十一个刻度、滑块拇指位于最左侧的滑块">

```tsx
import { useState } from 'react';
import { Host, Slider } from '@expo/ui/jetpack-compose';

export default function SteppedSliderExample() {
  const [value, setValue] = useState(0);

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <Slider
        value={value}
        min={0}
        max={100}
        steps={10}
        onValueChange={setValue}
      />
    </Host>
  );
}
```

</ComponentExample>

### 自定义颜色

使用 `colors` 属性覆盖滑块拇指、轨道和刻度的默认 Material3 颜色。

<ComponentExample
  title="CustomColorsSliderExample.tsx"
  src="/static/images/expo-ui/examples/slider-custom-colors-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/slider-custom-colors-android-dark.webp"
  alt="一个带紫色拇指、紫色活动轨道和灰色非活动轨道的滑块">

```tsx
import { useState } from 'react';
import { Host, Slider } from '@expo/ui/jetpack-compose';

export default function CustomColorsSliderExample() {
  const [value, setValue] = useState(0.5);

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <Slider
        value={value}
        colors={{
          thumbColor: '#6200EE',
          activeTrackColor: '#6200EE',
          inactiveTrackColor: '#E0E0E0',
        }}
        onValueChange={setValue}
      />
    </Host>
  );
}
```

</ComponentExample>

### 自定义拇指和轨道

同时使用 `Slider.Thumb` 和 `Slider.Track` 插槽，可完全自定义滑块外观。

<ComponentExample
  title="FullyCustomSliderExample.tsx"
  src="/static/images/expo-ui/examples/slider-custom-thumb-track-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/slider-custom-thumb-track-android-dark.webp"
  alt="一个带紫色圆形拇指、轨道由紫色和灰色两部分组成的滑块">

```tsx
import { useState } from 'react';
import {
  Host,
  Slider,
  Shape,
  Row,
  Box,
} from '@expo/ui/jetpack-compose';
import {
  fillMaxWidth,
  height,
  weight,
  size,
  clip,
  background,
  Shapes,
} from '@expo/ui/jetpack-compose/modifiers';

export default function FullyCustomSliderExample() {
  const [value, setValue] = useState(0.5);

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <Slider value={value} onValueChange={setValue}>
        <Slider.Thumb>
          <Box
            modifiers={[
              size(24, 24),
              clip(Shapes.Circle),
              background('#6200EE'),
            ]}
          />
        </Slider.Thumb>
        <Slider.Track>
          <Row modifiers={[fillMaxWidth(), height(8)]}>
            <Shape.RoundedCorner
              color="#6200EE"
              cornerRadii={{ topStart: 4, bottomStart: 4 }}
              modifiers={[weight(Math.max(value, 0.01)), height(8)]}
            />
            <Shape.RoundedCorner
              color="#BDBDBD"
              cornerRadii={{ topEnd: 4, bottomEnd: 4 }}
              modifiers={[
                weight(Math.max(1 - value, 0.01)),
                height(8),
              ]}
            />
          </Row>
        </Slider.Track>
      </Slider>
    </Host>
  );
}
```

</ComponentExample>

### 垂直滑块

使用 `VerticalSlider` 沿垂直轨道选择单个值。设置 `reverseDirection`，可使值从下往上递增，而不是从上往下递增。

<ComponentExample
  title="VerticalSliderExample.tsx"
  src="/static/images/expo-ui/examples/slider-vertical-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/slider-vertical-android-dark.webp"
  alt="一个垂直滑块，滑块拇指位于中点，填充轨道位于其下方">

```tsx
import { useState } from 'react';
import { Host, VerticalSlider } from '@expo/ui/jetpack-compose';
import { height } from '@expo/ui/jetpack-compose/modifiers';

export default function VerticalSliderExample() {
  const [value, setValue] = useState(0.5);

  return (
    <Host matchContents>
      <VerticalSlider
        value={value}
        reverseDirection
        onValueChange={setValue}
        modifiers={[height(240)]}
      />
    </Host>
  );
}
```

</ComponentExample>
