## 用法

`Box` 会将子元素层叠在一起。使用 `contentAlignment` 在 Box 内定位这些子元素。

<ComponentExample
  title="BoxExample.tsx"
  src="/static/images/expo-ui/examples/box-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/box-basic-android-dark.webp"
  alt="一个 200 × 200 的浅灰色方块，中央显示文本 Centered in Box">

```tsx
import { Host, Box, Text } from '@expo/ui/jetpack-compose';
import {
  size,
  background,
} from '@expo/ui/jetpack-compose/modifiers';

export default function BoxExample() {
  return (
    <Host matchContents>
      <Box
        contentAlignment="center"
        modifiers={[size(200, 200), background('#E0E0E0')]}>
        <Text>Centered in Box</Text>
      </Box>
    </Host>
  );
}
```

</ComponentExample>
