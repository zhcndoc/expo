Expo UI Picker 与官方 SwiftUI [Picker API](https://developer.apple.com/documentation/swiftui/picker) 一致，并通过 [`pickerStyle`](modifiers/#pickerstylestyle) 修饰符支持所有 Picker 样式。

## 分段 Picker

<ComponentExample
  title="SegmentedPickerExample.tsx"
  src="/static/images/expo-ui/examples/picker-segmented-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/picker-segmented-ios-dark.webp"
  alt="一个分段控件，其中 Apple 处于选中状态，旁边是 Banana 和 Orange">

```tsx
import { useState } from 'react';
import { Host, Picker, Text } from '@expo/ui/swift-ui';
import { pickerStyle, tag } from '@expo/ui/swift-ui/modifiers';

const options = ['Apple', 'Banana', 'Orange'];

export default function SegmentedPickerExample() {
  const [selectedTag, setSelectedTag] = useState(options[0]);

  // 分段式和滚轮式 Picker 会拉伸至指定的宽度，因此在 `matchContents` 下会折叠。
  return (
    <Host style={{ flex: 1 }}>
      <Picker
        modifiers={[pickerStyle('segmented')]}
        label="选择一种水果"
        selection={selectedTag}
        onSelectionChange={selection => {
          setSelectedTag(selection);
        }}>
        {options.map(option => (
          <Text key={option} modifiers={[tag(option)]}>
            {option}
          </Text>
        ))}
      </Picker>
    </Host>
  );
}
```

</ComponentExample>
