## 用法

### 基本颜色选择器

<ComponentExample
  title="ColorPickerExample.tsx"
  src="/static/images/expo-ui/examples/colorpicker-basic-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/colorpicker-basic-ios-dark.webp"
  alt="系统颜色选择器面板在 Grid 标签页中打开，不透明度为 100%">

```tsx
import { useState } from 'react';
import { Host, ColorPicker } from '@expo/ui/swift-ui';

export default function ColorPickerExample() {
  const [color, setColor] = useState('#FF6347');

  // 颜色选择器会填满分配给它的宽度，因此 `matchContents` 会将其收起。
  return (
    <Host style={{ flex: 1 }}>
      <ColorPicker
        label="Select a color"
        selection={color}
        onSelectionChange={setColor}
      />
    </Host>
  );
}
```

</ComponentExample>

### 支持不透明度的颜色选择器

使用 `supportsOpacity` 属性，允许用户选择带有 Alpha 透明度的颜色。

<ComponentExample
  title="ColorPickerOpacityExample.tsx"
  src="/static/images/expo-ui/examples/colorpicker-opacity-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/colorpicker-opacity-ios-dark.webp"
  alt="系统颜色选择器面板在 Grid 标签页中打开，不透明度为 50%">

```tsx
import { useState } from 'react';
import { Host, ColorPicker } from '@expo/ui/swift-ui';

export default function ColorPickerOpacityExample() {
  const [color, setColor] = useState('#FF634780');

  return (
    <Host style={{ flex: 1 }}>
      <ColorPicker
        label="Select a color with opacity"
        selection={color}
        onSelectionChange={setColor}
        supportsOpacity
      />
    </Host>
  );
}
```

</ComponentExample>
