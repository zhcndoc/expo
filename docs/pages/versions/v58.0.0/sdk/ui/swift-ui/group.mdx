## 用法

### 基本分组

分组适用于一次为多个视图应用修饰符，或在不影响布局的情况下组织视图。

<ComponentExample
  title="BasicGroupExample.tsx"
  src="/static/images/expo-ui/examples/group-basic-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/group-basic-ios-dark.webp"
  alt="三行垂直排列的蓝色文本：第一项、第二项和第三项。">

```tsx
import { Group, Host, Text, VStack } from '@expo/ui/swift-ui';
import { foregroundStyle } from '@expo/ui/swift-ui/modifiers';

export default function BasicGroupExample() {
  return (
    <Host matchContents style={{ alignSelf: 'center' }}>
      <VStack spacing={8}>
        <Group modifiers={[foregroundStyle('blue')]}>
          <Text>First item</Text>
          <Text>Second item</Text>
          <Text>Third item</Text>
        </Group>
      </VStack>
    </Host>
  );
}
```

</ComponentExample>
