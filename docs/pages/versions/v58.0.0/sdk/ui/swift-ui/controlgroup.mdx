Expo UI ControlGroup 与官方 SwiftUI [ControlGroup API](https://developer.apple.com/documentation/swiftui/controlgroup) 相对应。放在 [`Menu`](menu) 中时，子元素会显示为一行紧凑的水平按钮。

## 用法

### 基本控制组

菜单中的控制组会显示为一行水平排列的图标按钮。

<ComponentExample
  title="BasicControlGroupExample.tsx"
  src="/static/images/expo-ui/examples/controlgroup-basic-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/controlgroup-basic-ios-dark.webp"
  alt="打开的菜单，其中在 Other Action 条目上方显示一行 Add、Favorite 和 Share 图标">

```tsx
import {
  Host,
  Menu,
  ControlGroup,
  Button,
} from '@expo/ui/swift-ui';

export default function BasicControlGroupExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Menu label="Options" systemImage="ellipsis.circle">
        <ControlGroup>
          <Button
            systemImage="plus"
            label="Add"
            onPress={() => console.log('Add')}
          />
          <Button
            systemImage="star"
            label="Favorite"
            onPress={() => console.log('Favorite')}
          />
          <Button
            systemImage="square.and.arrow.up"
            label="Share"
            onPress={() => console.log('Share')}
          />
        </ControlGroup>
        <Button
          label="Other Action"
          onPress={() => console.log('Other')}
        />
      </Menu>
    </Host>
  );
}
```

</ComponentExample>
