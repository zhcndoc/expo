## 用法

### 基本分隔线

<ComponentExample
  title="BasicDividerExample.tsx"
  src="/static/images/expo-ui/examples/divider-basic-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/divider-basic-ios-dark.webp"
  alt="一条水平线将文本 First section 和 Second section 分隔开">

```tsx
import { Host, Divider, VStack, Text } from '@expo/ui/swift-ui';

export default function BasicDividerExample() {
  // 分隔线会延伸至指定的宽度，因此 `matchContents` 会使其收缩。
  return (
    <Host style={{ flex: 1 }}>
      <VStack>
        <Text>First section</Text>
        <Divider />
        <Text>Second section</Text>
      </VStack>
    </Host>
  );
}
```

</ComponentExample>

### 列表中的分隔线

<ComponentExample
  title="DividerInListExample.tsx"
  src="/static/images/expo-ui/examples/divider-in-list-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/divider-in-list-ios-dark.webp"
  alt="Item 1 到 Item 4 依次排列，每对项目之间都有一条水平线">

```tsx
import { Host, Divider, VStack, Text } from '@expo/ui/swift-ui';

export default function DividerInListExample() {
  return (
    <Host style={{ flex: 1 }}>
      <VStack spacing={8}>
        <Text>Item 1</Text>
        <Divider />
        <Text>Item 2</Text>
        <Divider />
        <Text>Item 3</Text>
        <Divider />
        <Text>Item 4</Text>
      </VStack>
    </Host>
  );
}
```

</ComponentExample>

### 上下文菜单中的分隔线

分隔线通常用于分隔上下文菜单中的操作组。

<ComponentExample
  title="DividerInContextMenuExample.tsx"
  src="/static/images/expo-ui/examples/divider-context-menu-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/divider-context-menu-ios-dark.webp"
  alt="一个上下文菜单，其中 Edit 和 Duplicate 与红色的 Delete 选项之间有一条分隔线">

```tsx
import {
  Host,
  ContextMenu,
  Button,
  Text,
  Divider,
} from '@expo/ui/swift-ui';

export default function DividerInContextMenuExample() {
  return (
    <Host style={{ flex: 1 }}>
      <ContextMenu>
        <ContextMenu.Items>
          <Button
            label="Edit"
            onPress={() => console.log('Edit')}
          />
          <Button
            label="Duplicate"
            onPress={() => console.log('Duplicate')}
          />
          <Divider />
          <Button
            label="Delete"
            role="destructive"
            onPress={() => console.log('Delete')}
          />
        </ContextMenu.Items>
        <ContextMenu.Trigger>
          <Text>Long press me</Text>
        </ContextMenu.Trigger>
      </ContextMenu>
    </Host>
  );
}
```

</ComponentExample>
