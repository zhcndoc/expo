## 用法

### 连续滑块

默认范围为 `[0, 1]`。

<ComponentExample
  title="ContinuousSliderExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-slider-continuous-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-slider-continuous-android-dark.webp',
    alt: '滑块控件位于中间位置的 Material 3 滑块',
  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-slider-continuous-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-slider-continuous-ios-dark.webp',
    alt: '滑块滑块位于中间位置的滑块',
  }}>

```tsx
import { useState } from 'react';
import { Host, Slider } from '@expo/ui';

export default function ContinuousSliderExample() {
  const [value, setValue] = useState(0.5);

  return (
    <Host matchContents={{ vertical: true }} style={{ width: '100%' }}>
      <Slider value={value} onValueChange={setValue} />
    </Host>
  );
}
```

</ComponentExample>

### 带自定义范围的分段滑块

使用 `min`、`max` 和 `step` 限制滑块产生的值。

<ComponentExample
  title="SteppedSliderExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-slider-stepped-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-slider-stepped-android-dark.webp',
    alt: 'Volume 标签位于带刻度标记的分段 Material 3 滑块上方',
  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-slider-stepped-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-slider-stepped-ios-dark.webp',
    alt: 'Volume 标签位于设置为范围中间值的滑块上方',
  }}>

```tsx
import { useState } from 'react';
import { useColorScheme } from 'react-native';
import { Host, Column, Slider, Text } from '@expo/ui';

export default function SteppedSliderExample() {
  const [volume, setVolume] = useState(50);
  const colorScheme = useColorScheme();

  return (
    <Host matchContents={{ vertical: true }} style={{ width: '100%' }}>
      <Column spacing={8}>
        <Text textStyle={{ color: colorScheme === 'dark' ? '#FFFFFF' : '#000000' }}>
          {`Volume: ${volume}`}
        </Text>
        <Slider value={volume} onValueChange={setVolume} min={0} max={100} step={10} />
      </Column>
    </Host>
  );
}
```

</ComponentExample>
